===============================================================================
LLPT CODE NANO (1B) - GUIA COMPLETO DE COMANDOS (DESKTOP & MOBILE)
===============================================================================

Este documento contém todos os comandos necessários para preparar, treinar,
fine-tunear, converter e executar a inferência do modelo LLPT Code Nano (1B).

-------------------------------------------------------------------------------
1. COMANDOS PARA AMBIENTE DESKTOP (Linux, macOS, Windows/WSL2)
-------------------------------------------------------------------------------

[1.1 INSTALAÇÃO DE DEPENDÊNCIAS]
  # Instalar dependências Python (PyTorch, Transformers, Accelerate, PEFT)
  pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
  pip install transformers datasets accelerate peft bitsandbytes sentencepiece trl

  # Clonar e compilar llama.cpp (para conversão GGUF e inferência rápida)
  git clone https://github.com/ggerganov/llama.cpp
  cd llama.cpp && make -j

[1.2 TREINAMENTO E FINE-TUNING]
  # Opção A: Executar script nativo de treino (Fine-tuning / Pre-training)
  python3 models/code/llpt-code-nano/src/train.py \
    --config models/code/llpt-code-nano/config.json \
    --data_path models/code/llpt-code-nano/dataset/train.jsonl \
    --output_dir models/code/llpt-code-nano/weights/checkpoints/ \
    --epochs 3 \
    --batch_size 4 \
    --learning_rate 2e-5

  # Opção B: Fine-Tuning eficiente com LoRA (Economiza memória VRAM)
  python3 models/code/llpt-code-nano/src/train.py \
    --config models/code/llpt-code-nano/config.json \
    --data_path models/code/llpt-code-nano/dataset/train.jsonl \
    --output_dir models/code/llpt-code-nano/weights/checkpoints/ \
    --use_lora

[1.3 CONVERSÃO E QUANTIZAÇÃO PARA GGUF]
  # Converter modelo Safetensors/PyTorch para GGUF FP16
  python3 llama.cpp/convert_hf_to_gguf.py models/code/llpt-code-nano/weights/final/ \
    --outtype f16 \
    --outfile models/code/llpt-code-nano/weights/gguf/llpt-code-nano-f16.gguf

  # Quantizar para Q4_K_M (Ideal para Mobile e Desktop leve)
  ./llama.cpp/llama-quantize models/code/llpt-code-nano/weights/gguf/llpt-code-nano-f16.gguf \
    models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf Q4_K_M

  # Quantizar para Q8_0 (Ideal para alta precisão em Desktop)
  ./llama.cpp/llama-quantize models/code/llpt-code-nano/weights/gguf/llpt-code-nano-f16.gguf \
    models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q8_0.gguf Q8_0

[1.4 INFERÊNCIA E TESTE NO DESKTOP]
  # Executar via llama.cpp no terminal (Modo Interativo / CLI)
  ./llama.cpp/llama-cli -m models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf \
    -p "def calculate_fibonacci(n):" -n 256

  # Subir servidor local compatível com API OpenAI para autocomplete em IDEs (VS Code)
  ./llama.cpp/llama-server -m models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf \
    --port 8080 --host 0.0.0.0


-------------------------------------------------------------------------------
2. COMANDOS PARA AMBIENTE MOBILE (Android via Termux & iOS)
-------------------------------------------------------------------------------

[2.1 CONFIGURAÇÃO DO AMBIENTE MOBILE (Android / Termux)]
  # Atualizar pacotes do Termux e instalar dependências essenciais
  pkg update && pkg upgrade -y
  pkg install git cmake clang python build-essential -y

  # Clonar e compilar llama.cpp nativamente no Android
  git clone https://github.com/ggerganov/llama.cpp
  cd llama.cpp
  mkdir build && cd build
  cmake ..
  cmake --build . --config Release -j4

[2.2 TREINAMENTO / FINE-TUNING LEVE NO MOBILE (Termux)]
  # Instalar PyTorch e Hugging Face no Termux (se necessário ajustar via Python)
  pip install transformers datasets peft

  # Executar ajuste fino leve via LoRA em CPU Mobile
  python3 models/code/llpt-code-nano/src/train.py \
    --config models/code/llpt-code-nano/config.json \
    --data_path models/code/llpt-code-nano/dataset/train.jsonl \
    --output_dir models/code/llpt-code-nano/weights/checkpoints/ \
    --epochs 1 \
    --batch_size 1 \
    --use_lora

[2.3 INFERÊNCIA NO MOBILE]
  # Executar o modelo quantizado Q4_K_M no smartphone via CLI
  ./llama-cli -m ~/models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf \
    -p "function formatResponse(data) {" -n 128 -t 4

  # Subir servidor HTTP leve no smartphone para servir de backend de código
  ./llama-server -m ~/models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf \
    --host 127.0.0.1 --port 8080 -t 4
