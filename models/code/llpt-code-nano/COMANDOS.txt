===============================================================================
LLPT CODE NANO (1B) - GUIA COMPLETO DE COMANDOS (DESKTOP & MOBILE)
===============================================================================

Este documento contém todos os comandos necessários para preparar, treinar com datasets .txt,
retomar, interromper, converter e executar a inferência do modelo LLPT Code Nano (1B).

-------------------------------------------------------------------------------
1. COMANDOS PARA AMBIENTE DESKTOP (Linux, macOS, Windows/WSL2)
-------------------------------------------------------------------------------

[1.1 INSTALAÇÃO DE DEPENDÊNCIAS]
  # Instalar dependências de PyTorch e Hugging Face
  pip install torch torchvision torchaudio
  pip install transformers datasets accelerate peft safetensors

  # Clonar e compilar llama.cpp (para conversão GGUF e inferência ultra-rápida C++)
  git clone https://github.com/ggerganov/llama.cpp
  cd llama.cpp && make -j

[1.2 INICIAR TREINAMENTO COM DATASET .TXT CATEGORIZADO]
  # Treinar com todo o diretório de datasets (Programação/ e Conversação/)
  python3 models/code/llpt-code-nano/src/train.py \
    --config models/code/llpt-code-nano/config.json \
    --data_path models/code/llpt-code-nano/dataset/ \
    --output_dir models/code/llpt-code-nano/weights/checkpoints/ \
    --epochs 3 \
    --batch_size 2 \
    --learning_rate 5e-5 \
    --max_length 512 \
    --prog_ratio 0.8

  # Treinar apenas com a pasta de programação ou arquivo .txt específico (ex: Programação/programacao.txt)
  python3 models/code/llpt-code-nano/src/train.py \
    --config models/code/llpt-code-nano/config.json \
    --data_path models/code/llpt-code-nano/dataset/Programação/programacao.txt \
    --output_dir models/code/llpt-code-nano/weights/checkpoints/ \
    --epochs 3

[1.3 RETOMAR TREINAMENTO A PARTIR DE UM CHECKPOINT]
  python3 models/code/llpt-code-nano/src/train.py \
    --config models/code/llpt-code-nano/config.json \
    --data_path models/code/llpt-code-nano/dataset/ \
    --output_dir models/code/llpt-code-nano/weights/checkpoints/ \
    --resume_from_checkpoint models/code/llpt-code-nano/weights/checkpoints/checkpoint-epoch-1 \
    --epochs 3

[1.4 INTERROMPER TREINAMENTO]
  # O script salva checkpoints a cada época em weights/checkpoints/checkpoint-epoch-X/
  # Você pode interromper o treinamento com Ctrl+C no terminal a qualquer momento
  # e retomar posteriormente usando --resume_from_checkpoint.

[1.5 INFERÊNCIA E TESTE NO DESKTOP]
  # Inferência em modo de continuação de código
  python3 models/code/llpt-code-nano/src/inference.py \
    --model_dir models/code/llpt-code-nano/weights/checkpoints/final/ \
    --prompt "def calculate_factorial(n):" \
    --max_tokens 64

  # Inferência em modo Fill-In-the-Middle (FIM)
  python3 models/code/llpt-code-nano/src/inference.py \
    --model_dir models/code/llpt-code-nano/weights/checkpoints/final/ \
    --prefix "def add(a, b):\n    " \
    --suffix "\n\ndef multiply(a, b):" \
    --max_tokens 32

[1.6 CONVERSÃO E QUANTIZAÇÃO PARA GGUF]
  # Converter modelo para GGUF FP16
  python3 llama.cpp/convert_hf_to_gguf.py models/code/llpt-code-nano/weights/checkpoints/final/ \
    --outtype f16 \
    --outfile models/code/llpt-code-nano/weights/gguf/llpt-code-nano-f16.gguf

  # Quantizar para Q4_K_M (Ideal para Mobile e Desktop leve)
  ./llama.cpp/llama-quantize models/code/llpt-code-nano/weights/gguf/llpt-code-nano-f16.gguf \
    models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf Q4_K_M

  # Executar via llama.cpp
  ./llama.cpp/llama-cli -m models/code/llpt-code-nano/weights/gguf/llpt-code-nano-q4_k_m.gguf \
    -p "def calculate_fibonacci(n):" -n 128
