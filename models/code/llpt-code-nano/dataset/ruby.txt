require 'json'
require 'time'
require 'securerandom'

# ==========================================
# 1. Modules & Mixins
# ==========================================

module Loggable
  def log_info(message)
    puts "[INFO] [#{Time.now.iso8601}] [#{self.class.name}]: #{message}"
  end

  def log_error(message, error = nil)
    err_str = error ? " (#{error.class}: #{error.message})" : ""
    puts "[ERROR] [#{Time.now.iso8601}] [#{self.class.name}]: #{message}#{err_str}"
  end
end

module Serializable
  def to_json(*args)
    as_json.to_json(*args)
  end

  def as_json
    instance_variables.each_with_object({}) do |var, hash|
      key = var.to_s.delete('@')
      hash[key] = instance_variable_get(var)
    end
  end
end

# ==========================================
# 2. Data Models & Validations
# ==========================================

class User
  include Loggable
  include Serializable

  ROLES = %i[admin developer guest].freeze

  attr_reader :id, :username, :email, :role, :created_at
  attr_accessor :active

  def initialize(username:, email:, role: :guest, active: true)
    @id = SecureRandom.uuid
    @username = username
    @email = validate_email!(email)
    @role = validate_role!(role)
    @active = active
    @created_at = Time.now.utc

    log_info("User instance created: #{@username} <#{@email}>")
  end

  def active?
    @active
  end

  def admin?
    @role == :admin
  end

  def update_role(new_role)
    @role = validate_role!(new_role)
    log_info("Role updated to '#{@role}' for user '#{@username}'")
  end

  private

  def validate_email!(email)
    raise ArgumentError, "Invalid email format: #{email}" unless email.match?(URI::MailTo::EMAIL_REGEXP)
    email.downcase
  end

  def validate_role!(role)
    symbolic_role = role.to_sym
    raise ArgumentError, "Invalid role: #{role}" unless ROLES.include?(symbolic_role)
    symbolic_role
  end
end

# ==========================================
# 3. Repository & Metaprogramming
# ==========================================

class UserRepository
  include Loggable

  def initialize
    @users = {}
  end

  def add(user)
    @users[user.id] = user
    log_info("Saved user ID: #{user.id}")
  end

  def find_by_id(id)
    @users[id] || raise("User not found with ID #{id}")
  end

  def find_active_by_role(role)
    @users.values.select { |u| u.active? && u.role == role.to_sym }
  end

  # Metaprogramming example: dynamic finder methods like find_by_username
  def method_missing(method_name, *args, &block)
    if method_name.to_s.start_with?('find_by_')
      attribute = method_name.to_s.sub('find_by_', '')
      value = args.first
      @users.values.find { |u| u.send(attribute) == value if u.respond_to?(attribute) }
    else
      super
    end
  end

  def respond_to_missing?(method_name, include_private = false)
    method_name.to_s.start_with?('find_by_') || super
  end
end
