public with sharing class AccountService {

    public class AccountException extends Exception {}

    public static List<Account> getActiveAccountsWithRecentOrders(Set<Id> accountIds) {
        if (accountIds == null || accountIds.isEmpty()) {
            return new List<Account>();
        }

        return [
            SELECT Id, Name, Rating, AnnualRevenue, Phone,
                   (SELECT Id, Amount, StageName, CloseDate FROM Opportunities WHERE IsClosed = false)
            FROM Account
            WHERE Id IN :accountIds AND Active__c = 'Yes'
            WITH SECURITY_ENFORCED
            ORDER BY Name ASC
        ];
    }

    public static void updateAccountRatingBulk(List<Account> accounts) {
        List<Account> accountsToUpdate = new List<Account>();

        for (Account acc : accounts) {
            if (acc.AnnualRevenue != null && acc.AnnualRevenue >= 1000000) {
                acc.Rating = 'Hot';
                accountsToUpdate.add(acc);
            } else if (acc.AnnualRevenue != null && acc.AnnualRevenue >= 100000) {
                acc.Rating = 'Warm';
                accountsToUpdate.add(acc);
            }
        }

        if (!accountsToUpdate.isEmpty()) {
            SObjectAccessDecision decision = Security.stripInaccessible(AccessType.UPDATABLE, accountsToUpdate);
            update decision.getRecords();
        }
    }
}

// Apex Trigger
trigger AccountTrigger on Account (before insert, before update) {
    if (Trigger.isBefore) {
        AccountService.updateAccountRatingBulk(Trigger.new);
    }
}
