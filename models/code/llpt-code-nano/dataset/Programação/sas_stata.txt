LIBNAME mydata '/data/llpt/sales';

DATA work.active_customers;
    SET mydata.customers;
    WHERE status = 'Active' AND annual_revenue >= 10000;

    IF tenure_years >= 5 THEN loyalty_tier = 'Gold';
    ELSE IF tenure_years >= 2 THEN loyalty_tier = 'Silver';
    ELSE loyalty_tier = 'Bronze';

    monthly_revenue = annual_revenue / 12;
    FORMAT monthly_revenue DOLLAR12.2;
RUN;

PROC MEANS DATA=work.active_customers N MEAN STD MIN MAX;
    CLASS loyalty_tier;
    VAR monthly_revenue tenure_years;
RUN;

PROC REG DATA=work.active_customers;
    MODEL annual_revenue = tenure_years total_orders;
RUN;
