% ==========================================
% 1. Facts and Rules
% ==========================================

parent(albert, bob).
parent(bob, charlie).

ancestor(X, Y) :- parent(X, Y).
ancestor(X, Y) :- parent(X, Z), ancestor(Z, Y).

factorial(0, 1).
factorial(N, F) :-
    N > 0,
    N1 is N - 1,
    factorial(N1, F1),
    F is N * F1.

append([], L, L).

append([H|T], L, [H|R]) :-
    append(T, L, R).
