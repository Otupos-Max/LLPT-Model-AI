program OberonDemo;

var
    counter: integer;

function CalculateFactorial(n: integer): integer;
begin
    if n <= 1 then
        CalculateFactorial := 1
    else
        CalculateFactorial := n * CalculateFactorial(n - 1);
end;

begin
    writeln('Hello from Oberon!');
    counter := CalculateFactorial(5);
end.
