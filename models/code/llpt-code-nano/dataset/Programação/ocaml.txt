module type STORAGE = sig
  type 'a t
  val create : unit -> 'a t
  val push : 'a -> 'a t -> unit
  val pop : 'a t -> 'a option
  val length : 'a t -> int
end

module Stack : STORAGE = struct
  type 'a t = 'a list ref

  let create () = ref []

  let push x s =
    s := x :: !s

  let pop s =
    match !s with
    | [] -> None
    | x :: xs ->
        s := xs;
        Some x

  let length s = List.length !s
end

(* Algebraic Data Types & GADTs *)
type _ expr =
  | Int : int -> int expr
  | Bool : bool -> bool expr
  | Add : int expr * int expr -> int expr
  | Eq : 'a expr * 'a expr -> bool expr

let rec eval : type a. a expr -> a = function
  | Int n -> n
  | Bool b -> b
  | Add (e1, e2) -> (eval e1) + (eval e2)
  | Eq (e1, e2) -> (eval e1) = (eval e2)

(* Tail Recursive Pipeline *)
let filter_map_tail f lst =
  let rec aux acc = function
    | [] -> List.rev acc
    | x :: xs ->
        match f x with
        | None -> aux acc xs
        | Some y -> aux (y :: acc) xs
  in aux [] lst
