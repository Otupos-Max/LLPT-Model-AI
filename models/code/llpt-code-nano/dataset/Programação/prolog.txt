% Facts & Rules
parent(albert, bob).

parent(bob, carl).
parent(bob, charlie).

ancestor(X, Y) :- parent(X, Y).
ancestor(X, Y) :- parent(X, Z), ancestor(Z, Y).

% Graph Path Finder with Loop Avoidance
path(Node, Node, _, [Node]).
path(Start, End, Visited, [Start|Path]) :-
    edge(Start, Next),
    \+ member(Next, Visited),
    path(Next, End, [Next|Visited], Path).

edge(a, b). edge(b, c). edge(c, d). edge(a, c).

% N-Queens Problem using Constraint Logic Programming over Finite Domains
n_queens(N, Qs) :-
    length(Qs, N),
    Qs ins 1..N,
    all_distinct(Qs),
    safe_queens(Qs),
    label(Qs).

safe_queens([]).
safe_queens([Q|Qs]) :-
    safe_queens(Qs, Q, 1),
    safe_queens(Qs).

safe_queens([], _, _).
safe_queens([Q|Qs], Q0, D0) :-
    Q0 #\= Q,
    abs(Q0 - Q) #\= D0,
    D1 #= D0 + 1,
    safe_queens(Qs, Q0, D1).
