# Example 2: Functions and Loops in fish
process_items() {
    local items=("$@")
    for item in "${items[@]}"; do
        echo "Processing item: $item"
    done
}

# ==========================================
# 2. Fill-In-The-Middle (FIM) Example
# ==========================================
<PRE>
#!/usr/bin/env fish
calculate_sum() {
    local total=0
<SUF>
    echo $total
}
<MID>
    for num in "$@"; do
        total=$((total + num))
    done
<EOT>
