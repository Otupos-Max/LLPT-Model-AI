package com.llpt.code.nano

import groovy.transform.Canonical
import groovy.transform.ToString
import groovy.transform.CompileStatic

@CompileStatic
@Canonical
class Person {
    String firstName
    String lastName
    int age
    List<String> hobbies = []

    String getFullName() {
        return "${firstName} ${lastName}"
    }
}

class GroovyDataProcessor {

    static List<Map<String, Object>> processUsers(List<Person> people) {
        return people
            .findAll { it.age >= 18 }
            .collect { [name: it.fullName, age: it.age, isAdult: true] }
    }

    static void metaProgrammingDemo() {
        String.metaClass.toSlug = { ->
            delegate.toLowerCase().replaceAll(/[^a-z0-9]+/, '-')
        }

        def title = "LLPT Code Nano 1B Model release!"
        println title.toSlug() // Output: llpt-code-nano-1b-model-release
    }
}

// Spock Specification Example
class PersonSpec extends spock.lang.Specification {

    def "should correctly process adult persons"() {
        given: "a list of people"
        def people = [
            new Person(firstName: "Alice", lastName: "Smith", age: 25),
            new Person(firstName: "Bob", lastName: "Junior", age: 15)
        ]

        when: "processing people list"
        def results = GroovyDataProcessor.processUsers(people)

        then: "only adults are returned"
        results.size() == 1
        results[0].name == "Alice Smith"
    }
}
