/* ==========================================
 * 1. Dynamic Array (Vector) Implementation
 * ========================================== */

typedef struct {
    int *data;
    size_t size;
    size_t capacity;
} Vector;

Vector *vector_create(size_t initial_capacity) {
    Vector *vec = (Vector *)malloc(sizeof(Vector));
    if (!vec) return NULL;

    vec->data = (int *)malloc(initial_capacity * sizeof(int));
    if (!vec->data) {
        free(vec);
        return NULL;
    }

    vec->size = 0;
    vec->capacity = initial_capacity;
    return vec;
}

bool vector_push(Vector *vec, int value) {
    if (!vec) return false;

    if (vec->size >= vec->capacity) {
        size_t new_capacity = vec->capacity * 2;
        int *new_data = (int *)realloc(vec->data, new_capacity * sizeof(int));
        if (!new_data) return false;

        vec->data = new_data;
        vec->capacity = new_capacity;
    }

    vec->data[vec->size++] = value;
    return true;
}

int vector_get(const Vector *vec, size_t index, bool *out_success) {
    if (!vec || index >= vec->size) {
        if (out_success) *out_success = false;
        return 0;
    }
    if (out_success) *out_success = true;
    return vec->data[index];
}

void vector_free(Vector *vec) {
    if (vec) {
        free(vec->data);
        free(vec);
    }
}

/* ==========================================
 * 2. Doubly Linked List
 * ========================================== */

typedef struct Node {
    int data;
    struct Node *prev;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    Node *tail;
    size_t length;
} LinkedList;

LinkedList *list_create(void) {
    LinkedList *list = (LinkedList *)malloc(sizeof(LinkedList));
    if (!list) return NULL;
    list->head = NULL;
    list->tail = NULL;
    list->length = 0;
    return list;
}

bool list_append(LinkedList *list, int data) {
    if (!list) return false;

    Node *new_node = (Node *)malloc(sizeof(Node));
    if (!new_node) return false;

    new_node->data = data;
    new_node->next = NULL;
    new_node->prev = list->tail;

    if (list->tail) {
        list->tail->next = new_node;
    } else {
        list->head = new_node;
    }

    list->tail = new_node;
    list->length++;
    return true;
}

void list_free(LinkedList *list) {
    if (!list) return;

    Node *current = list->head;
    while (current) {
        Node *next = current->next;
        free(current);
        current = next;
    }
    free(list);
}

/* ==========================================
 * 3. Hash Table with Chaining
 * ========================================== */

#define HASH_TABLE_SIZE 128

typedef struct Entry {
    char *key;
    char *value;
    struct Entry *next;
} Entry;

typedef struct {
    Entry *buckets[HASH_TABLE_SIZE];
} HashTable;

unsigned int hash(const char *key) {
    unsigned int value = 0;
    for (; *key != '\0'; key++) {
        value = value * 37 + (unsigned char)(*key);
    }
    return value % HASH_TABLE_SIZE;
}

HashTable *ht_create(void) {
    HashTable *ht = (HashTable *)calloc(1, sizeof(HashTable));
    return ht;
}

bool ht_put(HashTable *ht, const char *key, const char *value) {
    if (!ht || !key || !value) return false;

    unsigned int slot = hash(key);
    Entry *entry = ht->buckets[slot];

    while (entry) {
        if (strcmp(entry->key, key) == 0) {
            free(entry->value);
            entry->value = strdup(value);
            return true;
        }
        entry = entry->next;
    }

    Entry *new_entry = (Entry *)malloc(sizeof(Entry));
    if (!new_entry) return false;

    new_entry->key = strdup(key);
    new_entry->value = strdup(value);
    new_entry->next = ht->buckets[slot];
    ht->buckets[slot] = new_entry;
    return true;
}

const char *ht_get(HashTable *ht, const char *key) {
    if (!ht || !key) return NULL;

    unsigned int slot = hash(key);
    Entry *entry = ht->buckets[slot];

    while (entry) {
        if (strcmp(entry->key, key) == 0) {
            return entry->value;
        }
        entry = entry->next;
    }
    return NULL;
}

void ht_free(HashTable *ht) {
    if (!ht) return;

    for (int i = 0; i < HASH_TABLE_SIZE; i++) {
        Entry *entry = ht->buckets[i];
        while (entry) {
            Entry *next = entry->next;
            free(entry->key);
            free(entry->value);
            free(entry);
            entry = next;
        }
    }
    free(ht);
}

int main(void) {
    printf("--- C System Programming Examples ---\n");

    Vector *v = vector_create(4);
    for (int i = 1; i <= 10; i++) {
        vector_push(v, i * 10);
    }
    printf("Vector size: %zu, capacity: %zu\n", v->size, v->capacity);
    vector_free(v);

    HashTable *ht = ht_create();
    ht_put(ht, "language", "C");
    ht_put(ht, "version", "C11");
    printf("Language: %s | Version: %s\n", ht_get(ht, "language"), ht_get(ht, "version"));
    ht_free(ht);

    return 0;
}


/* ==========================================
 * 4. Memory Pool Allocator & Circular Buffer
 * ========================================== */

#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <string.h>

#define BUFFER_SIZE 8

typedef struct {
    int data[BUFFER_SIZE];
    size_t head;
    size_t tail;
    size_t count;
} CircularBuffer;

void cb_init(CircularBuffer *cb) {
    cb->head = 0;
    cb->tail = 0;
    cb->count = 0;
}

bool cb_push(CircularBuffer *cb, int value) {
    if (cb->count == BUFFER_SIZE) return false; // Buffer full
    cb->data[cb->tail] = value;
    cb->tail = (cb->tail + 1) % BUFFER_SIZE;
    cb->count++;
    return true;
}

bool cb_pop(CircularBuffer *cb, int *value) {
    if (cb->count == 0) return false; // Buffer empty
    *value = cb->data[cb->head];
    cb->head = (cb->head + 1) % BUFFER_SIZE;
    cb->count--;
    return true;
}

 * 3. Instruction & System Programming Example
 * ========================================== */

/* Instruction: Implement a thread-safe Circular Buffer (FIFO) in C using mutexes and condition variables. */
/* Response: */

#include <pthread.h>

typedef struct {
    int *buffer;
    size_t head;
    size_t tail;
    size_t max_len;
    size_t count;
    pthread_mutex_t mutex;
    pthread_cond_t not_full;
    pthread_cond_t not_empty;
} CircularBuffer;

CircularBuffer *cb_init(size_t capacity) {
    CircularBuffer *cb = (CircularBuffer *)malloc(sizeof(CircularBuffer));
    if (!cb) return NULL;
    cb->buffer = (int *)malloc(capacity * sizeof(int));
    cb->head = 0;
    cb->tail = 0;
    cb->max_len = capacity;
    cb->count = 0;
    pthread_mutex_init(&cb->mutex, NULL);
    pthread_cond_init(&cb->not_full, NULL);
    pthread_cond_init(&cb->not_empty, NULL);
    return cb;
}

void cb_push(CircularBuffer *cb, int data) {
    pthread_mutex_lock(&cb->mutex);
    while (cb->count == cb->max_len) {
        pthread_cond_wait(&cb->not_full, &cb->mutex);
    }
    cb->buffer[cb->tail] = data;
    cb->tail = (cb->tail + 1) % cb->max_len;
    cb->count++;
    pthread_cond_signal(&cb->not_empty);
    pthread_mutex_unlock(&cb->mutex);
}

int cb_pop(CircularBuffer *cb) {
    pthread_mutex_lock(&cb->mutex);
    while (cb->count == 0) {
        pthread_cond_wait(&cb->not_empty, &cb->mutex);
    }
    int data = cb->buffer[cb->head];
    cb->head = (cb->head + 1) % cb->max_len;
    cb->count--;
    pthread_cond_signal(&cb->not_full);
    pthread_mutex_unlock(&cb->mutex);
    return data;
}

/* ==========================================
 * 2. Memory Management & Custom Memory Arena
 * ========================================== */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>

typedef struct {
    uint8_t *buffer;
    size_t capacity;
    size_t offset;
} MemoryArena;

MemoryArena* arena_create(size_t capacity) {
    MemoryArena *arena = (MemoryArena*)malloc(sizeof(MemoryArena));
    if (!arena) return NULL;
    arena->buffer = (uint8_t*)malloc(capacity);
    if (!arena->buffer) {
        free(arena);
        return NULL;
    }
    arena->capacity = capacity;
    arena->offset = 0;
    return arena;
}

void* arena_alloc(MemoryArena *arena, size_t size) {
    size_t aligned_size = (size + 7) & ~7; // 8-byte alignment
    if (arena->offset + aligned_size > arena->capacity) {
        return NULL; // Out of memory in arena
    }
    void *ptr = &arena->buffer[arena->offset];
    arena->offset += aligned_size;
    return ptr;
}

void arena_reset(MemoryArena *arena) {
    arena->offset = 0;
}

void arena_destroy(MemoryArena *arena) {
    if (arena) {
        free(arena->buffer);
        free(arena);
    }
}

 * 3. Lock-free Atomic Lock & Threading
 * ========================================== */

#include <stdatomic.h>

typedef struct {
    atomic_flag flag;
} Spinlock;

void spinlock_init(Spinlock *lock) {
    atomic_flag_clear(&lock->flag);
}

void spinlock_lock(Spinlock *lock) {
    while (atomic_flag_test_and_set(&lock->flag)) {
        // busy wait
    }
}

void spinlock_unlock(Spinlock *lock) {
    atomic_flag_clear(&lock->flag);
}
