module main

import json
import time

struct User {
pub mut:
	id        int
	name      string
	email     string
	is_active bool
	created_at time.Time
}

fn new_user(id int, name string, email string) !User {
	if name.len == 0 {
		return error('User name cannot be empty')
	}
	if !email.contains('@') {
		return error('Invalid email address')
	}
	return User{
		id: id
		name: name
		email: email
		is_active: true
		created_at: time.now()
	}
}

fn (u User) to_json() string {
	return json.encode(u)
}

fn process_users_concurrently(users []User) []string {
	mut results := []string{}
	mut threads := []thread string{}

	for u in users {
		threads << go fn (usr User) string {
			return 'Processed user: ${usr.name} <${usr.email}>'
		}(u)
	}

	for t in threads {
		results << t.wait()
	}

	return results
}

fn main() {
	u := new_user(1, 'V Lang Dev', 'vdev@example.com') or {
		println('Failed to create user: ${err}')
		return
	}
	println(u.to_json())
}
