%% ==========================================
%% LLPT Code Nano - Erlang OTP & Actor Concurrency
%% ==========================================

-module(llpt_worker_server).
-behaviour(gen_server).

%% API
-export([start_link/0, async_compute/1, sync_query/1, stop/0]).

%% gen_server callbacks
-export([init/1, handle_call/3, handle_cast/2, handle_info/2, terminate/2, code_change/3]).

-record(state, {
    count = 0 :: non_neg_integer(),
    active_jobs = [] :: list()
}).

%% API Functions
start_link() ->
    gen_server:start_link({local, ?MODULE}, ?MODULE, [], []).

async_compute(JobData) ->
    gen_server:cast(?MODULE__, {compute, JobData}).

sync_query(Key) ->
    gen_server:call(?MODULE__, {query, Key}).

stop() ->
    gen_server:call(?MODULE__, stop).

%% gen_server callbacks
init([]) ->
    process_flag(trap_exit, true),
    {ok, #state{count = 0, active_jobs = []}}.

handle_call({query, Key}, _From, State) ->
    Reply = {ok, Key, State#state.count},
    {reply, Reply, State};

handle_call(stop, _From, State) ->
    {stop, normal, ok, State};

handle_call(_Request, _From, State) ->
    {reply, {error, unknown_call}, State}.

handle_cast({compute, JobData}, #state{count = C, active_jobs = Jobs} = State) ->
    NewJobs = [JobData | Jobs],
    NewState = State#state{count = C + 1, active_jobs = NewJobs},
    {noreply, NewState};

handle_cast(_Msg, State) ->
    {noreply, State}.

handle_info({'EXIT', _Pid, reason_normal}, State) ->
    {noreply, State};

handle_info(_Info, State) ->
    {noreply, State}.

terminate(_Reason, _State) ->
    ok.

code_change(_OldVsn, State, _Extra) ->
    {ok, State}.
