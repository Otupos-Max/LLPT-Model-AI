module llpt.engine;

import std.stdio;
import std.algorithm;
import std.array;
import std.exception;
import std.conv;

struct Account {
    ulong id;
    string name;
    double balance = 0.0;
    bool isActive = true;

    void deposit(double amount) pure {
        enforce!ArgumentException(amount > 0, "Deposit amount must be > 0");
        balance += amount;
    }

    void withdraw(double amount) pure {
        enforce!ArgumentException(amount <= balance, "Insufficient funds");
        balance -= amount;
    }
}

// Compile-Time CTFE Function Execution
pure double calculateCompoundInterest(double principal, double rate, int years) {
    double total = principal;
    foreach (i; 0 .. years) {
        total += total * rate;
    }
    return total;
}

// Static assert evaluated at compile time
static assert(calculateCompoundInterest(100.0, 0.05, 1) == 105.0);

// Generic Template Ranges Processing
auto processAccounts(R)(R accountsRange) if (isInputRange!R) {
    return accountsRange
        .filter!(a => a.isActive && a.balance > 100.0)
        .map!(a => tuple(a.name, a.balance))
        .array;
}

void main() {
    auto acc1 = Account(1, "D Dev", 250.0, true);
    writeln("Account balance: ", acc1.balance);
}
