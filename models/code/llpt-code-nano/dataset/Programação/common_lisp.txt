(defpackage :llpt-nano
  (:use :cl)
  (:export #:user #:make-user #:user-id #:user-name #:user-balance
           #:process-transaction #:calculate-interest))

(in-package :llpt-nano)

;; CLOS (Common Lisp Object System) Classes
(defclass user ()
  ((id :initarg :id
       :reader user-id
       :type integer)
   (name :initarg :name
         :accessor user-name
         :type string)
   (balance :initarg :balance
            :accessor user-balance
            :initform 0.0
            :type float)))

;; Generic Functions & Methods
(defgeneric process-transaction (account amount type)
  (:documentation "Processes a deposit or withdrawal for a given CLOS user."))

(defmethod process-transaction ((acc user) (amount number) (type (eql :deposit)))
  (incf (user-balance acc) amount)
  (format t "Deposited ~A. New Balance: ~A~%" amount (user-balance acc))
  (user-balance acc))

(defmethod process-transaction ((acc user) (amount number) (type (eql :withdrawal)))
  (if (>= (user-balance acc) amount)
      (progn
        (decf (user-balance acc) amount)
        (format t "Withdrew ~A. New Balance: ~A~%" amount (user-balance acc))
        (user-balance acc))
      (error "Insufficient funds: required ~A, available ~A" amount (user-balance acc))))

;; Common Lisp Macros
(defmacro with-timing (&body body)
  `(let ((start-time (get-internal-real-time)))
     (prog1 (progn ,@body)
       (let ((elapsed (/ (- (get-internal-real-time) start-time)
                         internal-time-units-per-second)))
         (format t "Execution took ~,3F seconds~%" elapsed)))))

;; Higher-Order Functions & Iteration
(defun filter-and-map (predicate transform list)
  (loop for item in list
        when (funcall predicate item)
        collect (funcall transform item)))
