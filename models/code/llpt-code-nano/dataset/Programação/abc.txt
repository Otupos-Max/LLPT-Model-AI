HOW TO RETURN max_val list:
    PUT 0 IN max
    FOR item IN list:
        IF item > max:
            PUT item IN max
    RETURN max

HOW TO PROCESS numbers:
    WRITE "Enter total numbers:"
    READ n RAW
    PUT {} IN scores
    FOR i IN {1..n}:
        WRITE "Enter value ", i, ":"
        READ val RAW
        INSERT val IN scores
    WRITE "Maximum value found: ", max_val scores

HOW TO RETURN sum_list items:
    PUT 0 IN total
    FOR x IN items:
        PUT total + x IN total
    RETURN total
