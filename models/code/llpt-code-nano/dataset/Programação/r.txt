library(dplyr)
library(tidyr)
library(ggplot2)

# Data Processing & Transformation
process_sales_data <- function(df, min_revenue = 1000) {
  cleaned_df <- df %>%
    filter(!is.na(customer_id)) %>%
    mutate(
      total_revenue = quantity * unit_price,
      order_date = as.Date(order_date),
      year_month = format(order_date, "%Y-%m")
    ) %>%
    filter(total_revenue >= min_revenue) %>%
    group_by(year_month, category) %>%
    summarise(
      monthly_revenue = sum(total_revenue),
      total_orders = n(),
      avg_order_value = mean(total_revenue),
      .groups = "drop"
    )

  return(cleaned_df)
}

# Linear Regression Model & Diagnostics
fit_linear_model <- function(data) {
  model <- lm(monthly_revenue ~ total_orders + avg_order_value, data = data)
  summary_info <- summary(model)

  diagnostics <- list(
    r_squared = summary_info$r.squared,
    coefficients = summary_info$coefficients,
    residuals = residuals(model)
  )

  return(list(model = model, diagnostics = diagnostics))
}
