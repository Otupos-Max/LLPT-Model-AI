fn_factorial_gml = function(n) {
    if (n <= 1) return 1;
    return n * fn_factorial_gml(n - 1);
}

fn_process_array_gml = function(arr) {
    let sum = 0;
    for (let i = 0; i < arr.length; i++) {
        sum += arr[i];
    }
    return sum;
}

// 2. Data Filtering and Business Logic

fn_filter_positive_gml = function(items) {
    let result = [];
    for (let i = 0; i < items.length; i++) {
        if (items[i] > 0) {
            result.push(items[i]);
        }
    }
    return result;
}
