namespace LLPT.Quantum {

    open Microsoft.Quantum.Canon;
    open Microsoft.Quantum.Intrinsic;
    open Microsoft.Quantum.Measurement;

    /// # Summary
    /// Creates Quantum Superposition using Hadamard gate and measures qubit.
    operation GenerateRandomBit() : Result {
        use q = Qubit();
        H(q); // Hadamard gate puts qubit into superposition |+>
        let result = M(q); // Measure qubit
        Reset(q);
        return result;
    }

    /// # Summary
    /// Creates Quantum Entanglement (Bell State |Φ+>) between two qubits.
    operation CreateBellState() : (Result, Result) {
        use (q1, q2) = (Qubit(), Qubit());
        H(q1);
        CNOT(q1, q2); // Controlled-NOT creates entangled state (|00> + |11>) / sqrt(2)

        let m1 = M(q1);
        let m2 = M(q2);

        Reset(q1);
        Reset(q2);

        return (m1, m2);
    }
}
