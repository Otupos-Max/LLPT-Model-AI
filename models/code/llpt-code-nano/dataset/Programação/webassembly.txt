(module
  ;; Memory declaration (1 page = 64KB)
  (memory (export "memory") 1)

  ;; Exported Function: Vector Add (i32 arrays)
  (func (export "vector_add") (param $ptr1 i32) (param $ptr2 i32) (param $out_ptr i32) (param $len i32)
    (local $i i32)
    (local.set $i (i32.const 0))
    (block $break
      (loop $top
        (br_if $break (i32.ge_s (local.get $i) (local.get $len)))

        ;; Load value from ptr1 + i*4
        (i32.store
          (i32.add (local.get $out_ptr) (i32.mul (local.get $i) (i32.const 4)))
          (i32.add
            (i32.load (i32.add (local.get $ptr1) (i32.mul (local.get $i) (i32.const 4))))
            (i32.load (i32.add (local.get $ptr2) (i32.mul (local.get $i) (i32.const 4))))
          )
        )

        (local.set $i (i32.add (local.get $i) (i32.const 1)))
        (br $top)
      )
    )
  )

  ;; Exported Function: Factorial (Iterative)
  (func (export "factorial") (param $n i64) (result i64)
    (local $result i64)
    (local.set $result (i64.const 1))
    (block $done
      (loop $loop
        (br_if $done (i64.le_s (local.get $n) (i64.const 1)))
        (local.set $result (i64.mul (local.get $result) (local.get $n)))
        (local.set $n (i64.sub (local.get $n) (i64.const 1)))
        (br $loop)
      )
    )
    (local.get $result)
  )
)
