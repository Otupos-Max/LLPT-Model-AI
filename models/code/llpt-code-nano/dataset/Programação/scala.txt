package com.llpt.code.nano

import scala.concurrent.{Future, ExecutionContext}
import scala.util.{Success, Failure}

enum OrderStatus:
  case Pending
  case Processing(workerId: String)
  case Completed(transactionId: String)
  case Failed(reason: String)

case class Item(id: String, name: String, price: BigDecimal)
case class Order(id: String, items: List[Item], status: OrderStatus):
  def total: BigDecimal = items.map(_.price).sum

object OrderProcessor:

  extension (o: Order)
    def isEligibleForDiscount: Boolean = o.total > 100.0

  def processOrderAsync(order: Order)(using ec: ExecutionContext): Future[Order] =
    Future {
      if order.items.isEmpty then
        order.copy(status = OrderStatus.Failed("Empty order items"))
      else
        // Simulating async order execution
        val txId = s"TX-${java.util.UUID.randomUUID().toString.take(8)}"
        order.copy(status = OrderStatus.Completed(txId))
    }

// Pattern Matching & Algebraic Data Types
sealed trait FinancialTransaction
case class Deposit(amount: BigDecimal) extends FinancialTransaction
case class Withdrawal(amount: BigDecimal) extends FinancialTransaction
case class Transfer(amount: BigDecimal, targetAccount: String) extends FinancialTransaction

def evaluateTransaction(balance: BigDecimal, tx: FinancialTransaction): Either[String, BigDecimal] =
  tx match
    case Deposit(amt) if amt > 0 => Right(balance + amt)
    case Withdrawal(amt) if amt <= balance => Right(balance - amt)
    case Withdrawal(amt) => Left("Insufficient funds for withdrawal")
    case Transfer(amt, target) if amt <= balance => Right(balance - amt)
    case Transfer(_, _) => Left("Insufficient funds for transfer")
