classdef SignalFilter < handle
    properties (Access = private)
        SamplingRate (1,1) double
        CutoffFreq (1,1) double
        FilterOrder (1,1) double
    end

    methods
        function obj = SignalFilter(Fs, Fc, Order)
            obj.SamplingRate = Fs;
            obj.CutoffFreq = Fc;
            obj.FilterOrder = Order;
        end

        function filtered_signal = apply_butterworth(obj, input_signal)
            Wn = obj.CutoffFreq / (obj.SamplingRate / 2);
            [b, a] = butter(obj.FilterOrder, Wn, 'low');
            filtered_signal = filter(b, a, input_signal);
        end

        function [f, P1] = compute_fft(obj, signal)
            L = length(signal);
            Y = fft(signal);
            P2 = abs(Y / L);
            P1 = P2(1:floor(L/2)+1);
            P1(2:end-1) = 2 * P1(2:end-1);
            f = obj.SamplingRate * (0:(L/2)) / L;
        end
    end
end

% ODE Solver Function
function dydt = pendulum_ode(t, y, L, g)
    theta = y(1);
    omega = y(2);
    dydt = [omega; -(g/L) * sin(theta)];
end
