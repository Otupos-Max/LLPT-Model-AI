import std/[asyncdispatch, json, strutils, sequtils, tables]

type
  Status = enum
    Active, Pending, Suspended

  User* = object
    id*: int
    name*: string
    email*: string
    status*: Status

proc newCustomer*(id: int, name, email: string): User =
  User(id: id, name: name, email: email, status: Status.Active)

proc toJson*(u: User): JsonNode =
  %* {
    "id": u.id,
    "name": u.name,
    "email": u.email,
    "status": $u.status
  }

# Metaprogramming Macro
import std/macros

macro benchmark*(name: string, body: untyped): untyped =
  result = quote do:
    let start = cpuTime()
    `body`
    let elapsed = cpuTime() - start
    echo "Benchmark [", `name`, "]: ", elapsed, " seconds"

# Async Processing
proc fetchRemoteDataAsync(url: string): Future[string] {.async.} =
  await sleepAsync(50)
  return "{\"status\": \"ok\", \"url\": \"" & url & "\"}"
