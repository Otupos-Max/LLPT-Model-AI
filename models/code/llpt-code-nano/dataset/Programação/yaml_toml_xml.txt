apiVersion: apps/v1
kind: Deployment
metadata:
  name: llpt-code-nano-api
  namespace: production
  labels:
    app.kubernetes.io/name: llpt-code-nano
    app.kubernetes.io/tier: backend
spec:
  replicas: 3
  selector:
    matchLabels:
      app: llpt-code-nano
  template:
    metadata:
      labels:
        app: llpt-code-nano
    spec:
      containers:
        - name: nano-server
          image: llpt/code-nano-engine:v1.2.0
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 8080
          resources:
            requests:
              memory: "512Mi"
              cpu: "250m"
            limits:
              memory: "2Gi"
              cpu: "1000m"
          env:
            - name: MODEL_PATH
              value: "/models/1b-nano"
            - name: PROG_RATIO
              value: "0.85"
