module LLPTScientific

using LinearAlgebra
using Statistics

export MatrixSolver, solve_system, calculate_statistics

struct MatrixSolver{T<:AbstractFloat}
    A::Matrix{T}
    b::Vector{T}

    function MatrixSolver(A::Matrix{T}, b::Vector{T}) where T<:AbstractFloat
        size(A, 1) == size(A, 2) || error("Matrix A must be square")
        size(A, 1) == length(b) || error("Dimension mismatch between A and b")
        new{T}(A, b)
    end
end

# Multiple Dispatch
function solve_system(solver::MatrixSolver{T})::Vector{T} where T
    return solver.A \ solver.b
end

function calculate_statistics(data::AbstractArray{T})::NamedTuple where T<:Real
    m = mean(data)
    s = std(data)
    med = median(data)
    return (mean=m, std=s, median=med)
end

# Parallelization & Macro Vectorization
function fast_vector_dot(x::Vector{Float64}, y::Vector{Float64})::Float64
    @assert length(x) == length(y)
    acc = 0.0
    @simd for i in 1:length(x)
        @inbounds acc += x[i] * y[i]
    end
    return acc
end

end # module
