const std = @import("std");
const Allocator = std.mem.Allocator;
const testing = std.testing;

pub const UserStatus = enum {
    active,
    pending,
    suspended,
};

pub const User = struct {
    id: u64,
    name: []const u8,
    email: []const u8,
    status: UserStatus,

    pub fn format(
        self: User,
        comptime fmt: []const u8,
        options: std.fmt.FormatOptions,
        writer: anytype,
    ) !void {
        _ = fmt;
        _ = options;
        try writer.print("User[id={d}, name={s}, email={s}, status={s}]", .{
            self.id,
            self.name,
            self.email,
            @tagName(self.status),
        });
    }
};

pub const UserRegistry = struct {
    allocator: Allocator,
    users: std.ArrayList(User),

    pub fn init(allocator: Allocator) UserRegistry {
        return UserRegistry{
            .allocator = allocator,
            .users = std.ArrayList(User).init(allocator),
        };
    }

    pub fn deinit(self: *UserRegistry) void {
        for (self.users.items) |user| {
            self.allocator.free(user.name);
            self.allocator.free(user.email);
        }
        self.users.deinit();
    }

    pub fn register(self: *UserRegistry, id: u64, name: []const u8, email: []const u8) !void {
        const dup_name = try self.allocator.dupe(u8, name);
        errdefer self.allocator.free(dup_name);

        const dup_email = try self.allocator.dupe(u8, email);
        errdefer self.allocator.free(dup_email);

        const user = User{
            .id = id,
            .name = dup_name,
            .email = dup_email,
            .status = .active,
        };

        try self.users.append(user);
    }
};

// Comptime Generic Function
pub fn genericVectorSum(comptime T: type, slice: []const T) T {
    var sum: T = 0;
    for (slice) |item| {
        sum += item;
    }
    return sum;
}

test "UserRegistry registration test" {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const allocator = gpa.allocator();

    var registry = UserRegistry.init(allocator);
    defer registry.deinit();

    try registry.register(1, "Zig Engine", "zig@example.com");
    try testing.expectEqual(@as(usize, 1), registry.users.items.len);
    try testing.expectEqualStrings("Zig Engine", registry.users.items[0].name);
}
