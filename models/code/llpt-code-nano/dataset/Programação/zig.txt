const std = @import("std");
const Allocator = std.mem.Allocator;

pub const ArrayList = struct {
    items: []i32,
    capacity: usize,
    allocator: Allocator,

    pub fn init(allocator: Allocator) ArrayList {
        return ArrayList{
            .items = &[_]i32{},
            .capacity = 0,
            .allocator = allocator,
        };
    }

    pub fn deinit(self: *ArrayList) void {
        if (self.capacity > 0) {
            self.allocator.free(self.items);
        }
    }

    pub fn append(self: *ArrayList, item: i32) !void {
        if (self.items.len == self.capacity) {
            const new_cap = if (self.capacity == 0) 8 else self.capacity * 2;
            const new_buf = try self.allocator.realloc(self.items, new_cap);
            self.items = new_buf;
            self.capacity = new_cap;
        }
        self.items.len += 1;
        self.items[self.items.len - 1] = item;
    }
};


// Zig Memory Allocator and Error Union Pattern

const std = @import("std");

pub const Stack = struct {
    items: std.ArrayList(i32),

    pub fn init(allocator: std.mem.Allocator) Stack {
        return Stack{
            .items = std.ArrayList(i32).init(allocator),
        };
    }

    pub fn deinit(self: *Stack) void {
        self.items.deinit();
    }

    pub fn push(self: *Stack, item: i32) !void {
        try self.items.append(item);
    }

    pub fn pop(self: *Stack) ?i32 {
        return self.items.popOrNull();
    }
};
