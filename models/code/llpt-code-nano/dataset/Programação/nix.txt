# LLPT Code Nano - Nix Expression Language & NixOS Flakes
# ==========================================

{
  description = "LLPT Code Nano 1B AI model development environment";

  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-unstable";
    flake-utils.url = "github:numtide/flake-utils";
  };

  outputs = { self, nixpkgs, flake-utils }:
    flake-utils.lib.eachDefaultSystem (system:
      let
        pkgs = import nixpkgs {
          inherit system;
          config.allowUnfree = true;
        };

        pythonEnv = pkgs.python311.withPackages (ps: with ps; [
          torch
          transformers
          sentencepiece
          datasets
          peft
          accelerate
        ]);
      in
      {
        devShells.default = pkgs.mkShell {
          buildInputs = [
            pythonEnv
            pkgs.git
            pkgs.cmake
            pkgs.ninja
            pkgs.gcc
          ];

          shellHook = ''
            echo "=========================================="
            echo "Welcome to LLPT Code Nano Flake Environment!"
            echo "Python $(python3 --version)"
            echo "=========================================="
          '';
        };

        packages.default = pkgs.stdenv.mkDerivation {
          pname = "llpt-code-nano";
          version = "1.2.0";
          src = ./.;

          installPhase = ''
            mkdir -p $out/bin $out/share/llpt
            cp -r models/ $out/share/llpt/
          '';
        };
      });
}
