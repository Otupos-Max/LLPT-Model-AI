; ==========================================
; LLPT Code Nano - LLVM Intermediate Representation (IR)
; ==========================================

target datalayout = "e-m:e-p270:32:32-p271:32:32-p272:64:64-i64:64-f80:128-n8:16:32:64-S128"
target triple = "x86_64-unknown-linux-gnu"

%struct.Vector3 = type { float, float, float }

; Function: Add two floats
define float @add_floats(float %a, float %b) #0 {
entry:
  %res = fadd float %a, %b
  ret float %res
}

; Function: Factorial (Loop in LLVM IR)
define i64 @factorial(i64 %n) #0 {
entry:
  %cmp = icmp sle i64 %n, 1
  br i1 %cmp, label %if.then, label %if.else

if.then:
  ret i64 1

if.else:
  %n.sub = sub nsw i64 %n, 1
  %call = call i64 @factorial(i64 %n.sub)
  %mult = mul nsw i64 %n, %call
  ret i64 %mult
}

; Function: Vector3 Dot Product
define float @vector3_dot(%struct.Vector3* %v1, %struct.Vector3* %v2) #0 {
entry:
  %x1.ptr = getelementptr inbounds %struct.Vector3, %struct.Vector3* %v1, i32 0, i32 0
  %x1 = load float, float* %x1.ptr, align 4
  %x2.ptr = getelementptr inbounds %struct.Vector3, %struct.Vector3* %v2, i32 0, i32 0
  %x2 = load float, float* %x2.ptr, align 4
  %prod.x = fmul float %x1, %x2

  %y1.ptr = getelementptr inbounds %struct.Vector3, %struct.Vector3* %v1, i32 0, i32 1
  %y1 = load float, float* %y1.ptr, align 4
  %y2.ptr = getelementptr inbounds %struct.Vector3, %struct.Vector3* %v2, i32 0, i32 1
  %y2 = load float, float* %y2.ptr, align 4
  %prod.y = fmul float %y1, %y2

  %sum = fadd float %prod.x, %prod.y
  ret float %sum
}

attributes #0 = { nounwind sspreq uwtable }
