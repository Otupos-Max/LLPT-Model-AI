// Swift Dataset - Modern Swift Concurrency, Protocols, Enums & SwiftUI Architecture

import Foundation

// ==========================================
// 1. Enums with Associated Values & Protocols
// ==========================================

public enum UserRole: String, Codable, CaseIterable {
    case admin
    case developer
    case guest
}

public enum NetworkResult<T> {
    case success(T)
    case failure(Error)
}

public protocol IdentifiableEntity {
    var id: UUID { get }
    var createdAt: Date { get }
}

// ==========================================
// 2. Structs, Value Types & Codable
// ==========================================

public struct User: IdentifiableEntity, Codable, Equatable {
    public let id: UUID
    public var username: String
    public var email: String
    public var role: UserRole
    public var isActive: Bool
    public let createdAt: Date

    public init(
        id: UUID = UUID(),
        username: String,
        email: String,
        role: UserRole = .guest,
        isActive: Bool = true,
        createdAt: Date = Date()
    ) {
        self.id = id
        self.username = username
        self.email = email
        self.role = role
        self.isActive = isActive
        self.createdAt = createdAt
    }

    public mutating func deactivate() {
        self.isActive = false
    }
}

// Extension with Computed Properties & Utilities
extension User {
    public var isAdmin: Bool {
        return role == .admin
    }

    public var formattedSummary: String {
        return "[\(role.rawValue.uppercased())] \(username) (\(email)) - Active: \(isActive)"
    }
}

// Custom Error Type
public enum ServiceError: LocalizedError {
    case userNotFound(UUID)
    case invalidEmail(String)
    case networkFailure(String)

    public var errorDescription: String? {
        switch self {
        case .userNotFound(let id):
            return "User with ID \(id) could not be found."
        case .invalidEmail(let email):
            return "The provided email address '\(email)' is invalid."
        case .networkFailure(let reason):
            return "Network operation failed: \(reason)."
        }
    }
}

// ==========================================
// 3. Async/Await & Actor Concurrency
// ==========================================

public actor UserDatabaseStore {
    private var users: [UUID: User] = [:]

    public init() {}

    public func save(user: User) async throws {
        guard user.email.contains("@") else {
            throw ServiceError.invalidEmail(user.email)
        }
        users[user.id] = user
    }

    public func findUser(by id: UUID) async throws -> User {
        guard let user = users[id] else {
            throw ServiceError.userNotFound(id)
        }
        return user
    }

    public func fetchActiveDevelopers() async -> [User] {
        return users.values.filter { $0.isActive && $0.role == .developer }
            .sorted { $0.username < $1.username }
    }
}

// ==========================================
// 4. Usage Example
// ==========================================

@main
struct SwiftDatasetMain {
    static func main() async {
        print("=== Modern Swift Concurrency Examples ===")

        let store = UserDatabaseStore()
        let alice = User(username: "alice", email: "alice@swift.org", role: .admin)
        let bob = User(username: "bob", email: "bob@swift.org", role: .developer)

        do {
            try await store.save(user: alice)
            try await store.save(user: bob)

            let fetchedAlice = try await store.findUser(by: alice.id)
            print("Fetched User: \(fetchedAlice.formattedSummary)")

            let developers = await store.fetchActiveDevelopers()
            print("Active Developers Count: \(developers.count)")
        } catch {
            print("Execution failed with error: \(error.localizedDescription)")
        }
    }
}

// 4. Swift Concurrency & Actor Model

import Foundation

actor BankAccount {
    private(set) var balance: Double

    init(initialBalance: Double) {
        self.balance = initialBalance
    }

    func deposit(amount: Double) {
        balance += amount
    }

    func withdraw(amount: Double) throws {
        guard balance >= amount else {
            throw NSError(domain: "InsufficientFunds", code: 400)
        }
        balance -= amount
    }
}

actor BankAccount {
    private var balance: Double

    init(initialBalance: Double) {
        self.balance = initialBalance
    }

    func deposit(amount: Double) {
        balance += amount
    }

    func withdraw(amount: Double) throws -> Double {
        guard balance >= amount else {
            throw NSError(domain: "InsufficientFunds", code: 400)
        }
        balance -= amount
        return balance
    }
