# GitHub Actions Automated CI/CD Pipeline Workflow

name: Build, Test and Deploy LLPT Engine

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Install Dependencies
        run: |
          python -m pip install --upgrade pip
          pip install torch transformers pytest

      - name: Run Test Suite
        run: |
          python3 models/code/llpt-code-nano/tests/test_pipeline.py

  docker-build:
    needs: test
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Build Docker Image
        run: |
          docker build -t llpt-code-nano:latest .
# ==========================================
# Native GitHub Actions Artifact Upload
# ==========================================

    - name: Archive Production Artifacts
      uses: actions/upload-artifact@v4
      with:
        name: dist-files
        path: dist/
