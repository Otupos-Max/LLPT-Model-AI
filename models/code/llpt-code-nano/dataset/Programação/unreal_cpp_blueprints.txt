#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Character.h"
#include "LLPTCharacter.generated.h"

DECLARE_DYNAMIC_MULTICAST_DELEGATE_TwoParams(FOnHealthChanged, float, CurrentHealth, float, MaxHealth);

UCLASS()
class LLPTNANO_API ALLPTCharacter : public ACharacter
{
	GENERATED_BODY()

public:
	ALLPTCharacter();

protected:
	virtual void BeginPlay() override;

public:
	virtual void Tick(float DeltaTime) override;
	virtual void SetupPlayerInputComponent(class UInputComponent* PlayerInputComponent) override;

	UPROPERTY(EditAnywhere, BlueprintReadWrite, Category = "Attributes")
	float MaxHealth = 100.0f;

	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "Attributes")
	float CurrentHealth;

	UPROPERTY(BlueprintAssignable, Category = "Events")
	FOnHealthChanged OnHealthChanged;

	UFUNCTION(BlueprintCallable, Category = "Attributes")
	void ApplyDamage(float DamageAmount);
};

// Implementation
#include "LLPTCharacter.h"

ALLPTCharacter::ALLPTCharacter()
{
	PrimaryActorTick.bCanEverTick = true;
	CurrentHealth = MaxHealth;
}

void ALLPTCharacter::BeginPlay()
{
	Super::BeginPlay();
	CurrentHealth = MaxHealth;
	OnHealthChanged.Broadcast(CurrentHealth, MaxHealth);
}

void ALLPTCharacter::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
}

void ALLPTCharacter::SetupPlayerInputComponent(UInputComponent* PlayerInputComponent)
{
	Super::SetupPlayerInputComponent(PlayerInputComponent);
}

void ALLPTCharacter::ApplyDamage(float DamageAmount)
{
	if (DamageAmount <= 0.0f || CurrentHealth <= 0.0f) return;

	CurrentHealth = FMath::Max(0.0f, CurrentHealth - DamageAmount);
	OnHealthChanged.Broadcast(CurrentHealth, MaxHealth);

	if (CurrentHealth <= 0.0f) {
		// Handle Death
	}
}

// ==========================================
// FIM Examples
// ==========================================
<PRE>UFUNCTION(BlueprintPure, Category = "Attributes")
bool IsAlive() const {
<SUF>}
<MID>    return CurrentHealth > 0.0f;
<EOT>
