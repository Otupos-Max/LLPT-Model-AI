Option Explicit

Public Sub ProcessCustomerExcelReport()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("SalesData")

    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    Dim i As Long
    Dim totalRevenue As Double
    totalRevenue = 0#

    Application.ScreenUpdating = False

    For i = 2 To lastRow
        Dim quantity As Long
        Dim unitPrice As Double

        quantity = ws.Cells(i, 3).Value
        unitPrice = ws.Cells(i, 4).Value

        Dim lineTotal As Double
        lineTotal = quantity * unitPrice
        ws.Cells(i, 5).Value = lineTotal

        If lineTotal >= 1000# Then
            ws.Cells(i, 5).Interior.Color = RGB(144, 238, 144) ' Light green
        End If

        totalRevenue = totalRevenue + lineTotal
    Next i

    ws.Cells(lastRow + 2, 4).Value = "Total Revenue:"
    ws.Cells(lastRow + 2, 5).Value = totalRevenue
    ws.Cells(lastRow + 2, 5).Font.Bold = True

    Application.ScreenUpdating = True
    MsgBox "Excel Sales Report successfully processed! Total: " & FormatCurrency(totalRevenue), vbInformation
End Sub
