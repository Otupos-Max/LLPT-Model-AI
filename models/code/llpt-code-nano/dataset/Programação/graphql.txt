"""
Directives for authentication & rate limiting
"""
directive @auth(requires: Role = USER) on OBJECT | FIELD_DEFINITION
directive @rateLimit(max: Int = 60, window: String = "1m") on FIELD_DEFINITION

enum Role {
  GUEST
  USER
  ADMIN
}

enum OrderStatus {
  PENDING
  PROCESSING
  SHIPPED
  DELIVERED
  CANCELLED
}

interface Node {
  id: ID!
  createdAt: String!
  updatedAt: String!
}

type User implements Node {
  id: ID!
  createdAt: String!
  updatedAt: String!
  username: String!
  email: String! @auth(requires: ADMIN)
  role: Role!
  orders(first: Int = 10, after: String): OrderConnection!
}

type Product implements Node {
  id: ID!
  createdAt: String!
  updatedAt: String!
  name: String!
  description: String
  price: Float!
  sku: String!
}

type Order implements Node {
  id: ID!
  createdAt: String!
  updatedAt: String!
  customer: User!
  items: [OrderItem!]!
  totalAmount: Float!
  status: OrderStatus!
}

type OrderItem {
  product: Product!
  quantity: Int!
  unitPrice: Float!
}

type OrderConnection {
  edges: [OrderEdge!]!
  pageInfo: PageInfo!
}

type OrderEdge {
  cursor: String!
  node: Order!
}

type PageInfo {
  hasNextPage: Boolean!
  endCursor: String
}

input CreateOrderInput {
  customerId: ID!
  items: [OrderItemInput!]!
}

input OrderItemInput {
  productId: ID!
  quantity: Int!
}

type Query {
  me: User @auth(requires: USER)
  user(id: ID!): User
  products(limit: Int = 20, offset: Int = 0): [Product!]! @rateLimit(max: 100)
}

type Mutation {
  createOrder(input: CreateOrderInput!): Order! @auth(requires: USER)
  updateOrderStatus(orderId: ID!, status: OrderStatus!): Order! @auth(requires: ADMIN)
}

type Subscription {
  orderUpdated(orderId: ID!): Order!
}
