from typing import List

def safe_batch_access(batch: List[int], index: int, default: int = 0) -> int:
    """Retorna o item no índice se válido, caso contrário retorna o valor padrão seguro."""
    if 0 <= index < len(batch):
        return batch[index]
    return default

/**
 * Equivalent JS Implementation:
 * @param {number} userId
 * @returns {Promise<{id: number, status: string}>}
 */
async function fetchUserData(userId) {
    await new Promise(resolve => setTimeout(resolve, 100));
    return { id: userId, status: "active" };
}

import logging

def log_and_raise(err: Exception, msg: str):
    logger.error("%s: %s", msg, str(err))
    raise RuntimeError(msg) from err
