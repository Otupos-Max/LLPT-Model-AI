module linear_algebra_mod
    implicit none
    private
    public :: matrix_mult, vector_norm, solve_linear_system

contains

    pure function vector_norm(v) result(norm_val)
        real(kind=8), intent(in) :: v(:)
        real(kind=8) :: norm_val
        norm_val = sqrt(sum(v**2))
    end function vector_norm

    subroutine matrix_mult(A, B, C, n, m, p)
        integer, intent(in) :: n, m, p
        real(kind=8), intent(in) :: A(n, m), B(m, p)
        real(kind=8), intent(out) :: C(n, p)
        integer :: i, j, k

        ! OpenMP parallel loop for high performance
        !$omp parallel do collapse(2) private(i, j, k)
        do j = 1, p
            do i = 1, n
                C(i, j) = 0.0_8
                do k = 1, m
                    C(i, j) = C(i, j) + A(i, k) * B(k, j)
                end do
            end do
        end do
        !$omp end parallel do
    end subroutine matrix_mult

    subroutine solve_linear_system(A, b, x, n)
        integer, intent(in) :: n
        real(kind=8), intent(inout) :: A(n, n)
        real(kind=8), intent(inout) :: b(n)
        real(kind=8), intent(out) :: x(n)
        integer :: i, j, k, max_row
        real(kind=8) :: factor, pivot_val

        ! Gaussian elimination with partial pivoting
        do k = 1, n - 1
            max_row = k
            pivot_val = abs(A(k, k))
            do i = k + 1, n
                if (abs(A(i, k)) > pivot_val) then
                    pivot_val = abs(A(i, k))
                    max_row = i
                end if
            end do

            if (max_row /= k) then
                call swap_rows(A, b, k, max_row, n)
            end if

            do i = k + 1, n
                factor = A(i, k) / A(k, k)
                A(i, k:n) = A(i, k:n) - factor * A(k, k:n)
                b(i) = b(i) - factor * b(k)
            end do
        end do

        ! Back substitution
        x(n) = b(n) / A(n, n)
        do i = n - 1, 1, -1
            x(i) = (b(i) - sum(A(i, i+1:n) * x(i+1:n))) / A(i, i)
        end do
    contains
        subroutine swap_rows(Mat, vec, r1, r2, dim_val)
            integer, intent(in) :: r1, r2, dim_val
            real(kind=8), intent(inout) :: Mat(dim_val, dim_val), vec(dim_val)
            real(kind=8) :: temp_row(dim_val), temp_v
            temp_row = Mat(r1, :)
            Mat(r1, :) = Mat(r2, :)
            Mat(r2, :) = temp_row
            temp_v = vec(r1)
            vec(r1) = vec(r2)
            vec(r2) = temp_v
        end subroutine swap_rows
    end subroutine solve_linear_system

end module linear_algebra_mod
