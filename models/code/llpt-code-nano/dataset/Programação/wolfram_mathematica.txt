(* Functional Pipeline & Matrix Manipulations *)
ProcessDataset[data_List, minThreshold_Numeric] := Module[
    {cleanedData, aggregated, model},

    cleanedData = Select[data, #["Value"] >= minThreshold &];

    aggregated = GroupBy[cleanedData, #["Category"] &, Mean[#[[All, "Value"]]] &];

    model = LinearModelFit[
        Table[{i, cleanedData[[i, "Value"]]}, {i, Length[cleanedData]}],
        x, x
    ];

    <|
        "CleanedCount" -> Length[cleanedData],
        "AggregatedMeans" -> aggregated,
        "RSquared" -> model["RSquared"],
        "Formula" -> model["BestFit"]
    |>
]

(* Differential Equation Solver *)
SolveHarmonicOscillator[k_, m_, x0_, v0_, tMax_] := Module[
    {sol},
    sol = NDSolveValue[
        {m x''[t] + k x[t] == 0, x[0] == x0, x'[0] == v0},
        x, {t, 0, tMax}
    ];
    Plot[sol[t], {t, 0, tMax}, PlotLabel -> "Harmonic Oscillator Trajectory"]
]
