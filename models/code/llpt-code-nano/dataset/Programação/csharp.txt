using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

namespace LlptCode.Dataset
{
    // ==========================================
    // 1. Records, Enums & Interfaces
    // ==========================================

    public enum UserStatus
    {
        Active,
        Inactive,
        PendingApproval,
        Suspended
    }

    public record UserDto(
        Guid Id,
        string Username,
        string Email,
        UserStatus Status,
        DateTime CreatedAt
    );

    public record CreateUserRequest(string Username, string Email);

    public interface IUserService
    {
        Task<UserDto> GetByIdAsync(Guid id, CancellationToken ct = default);
        IAsyncEnumerable<UserDto> GetAllActiveUsersAsync(CancellationToken ct = default);
        Task<UserDto> CreateUserAsync(CreateUserRequest request, CancellationToken ct = default);
    }

    // ==========================================
    // 2. Extension Methods & Custom Exceptions
    // ==========================================

    public static class StringExtensions
    {
        public static bool IsValidEmail(this string email)
        {
            if (string.IsNullOrWhiteSpace(email)) return false;
            return email.Contains('@') && email.Contains('.');
        }
    }

    public class UserNotFoundException : Exception
    {
        public UserNotFoundException(Guid id) : base($"User with ID {id} was not found.") { }
    }

    // ==========================================
    // 3. Service Implementation with LINQ & Async
    // ==========================================

    public class UserService : IUserService
    {
        private readonly Dictionary<Guid, UserDto> _users = new();

        public UserService()
        {
            var u1 = new UserDto(Guid.NewGuid(), "alice", "alice@example.com", UserStatus.Active, DateTime.UtcNow);
            var u2 = new UserDto(Guid.NewGuid(), "bob", "bob@example.com", UserStatus.Active, DateTime.UtcNow);
            _users[u1.Id] = u1;
            _users[u2.Id] = u2;
        }

        public async Task<UserDto> GetByIdAsync(Guid id, CancellationToken ct = default)
        {
            await Task.Delay(50, ct); // Simulate IO operation
            if (_users.TryGetValue(id, out var user))
            {
                return user;
            }
            throw new UserNotFoundException(id);
        }

        public async IAsyncEnumerable<UserDto> GetAllActiveUsersAsync(CancellationToken ct = default)
        {
            var activeUsers = _users.Values
                .Where(u => u.Status == UserStatus.Active)
                .OrderBy(u => u.Username);

            foreach (var user in activeUsers)
            {
                await Task.Delay(20, ct);
                yield return user;
            }
        }

        public async Task<UserDto> CreateUserAsync(CreateUserRequest request, CancellationToken ct = default)
        {
            if (!request.Email.IsValidEmail())
            {
                throw new ArgumentException("Invalid email format", nameof(request.Email));
            }

            await Task.Delay(100, ct);
            var newUser = new UserDto(
                Guid.NewGuid(),
                request.Username,
                request.Email,
                UserStatus.Active,
                DateTime.UtcNow
            );

            _users[newUser.Id] = newUser;
            return newUser;
        }

        // Pattern Matching Example
        public string EvaluateUserStatus(UserDto user) => user.Status switch
        {
            UserStatus.Active => "User is fully operational.",
            UserStatus.PendingApproval => "Awaiting administrator validation.",
            UserStatus.Inactive or UserStatus.Suspended => "Access restricted.",
            _ => "Unknown status code."
        };
    }

    // ==========================================
    // 4. Console App Entrypoint
    // ==========================================

    public class Program
    {
        public static async Task Main(string[] args)
        {
            Console.WriteLine("=== C# .NET 8 Dataset Examples ===");

            IUserService userService = new UserService();

            var created = await userService.CreateUserAsync(new CreateUserRequest("charlie", "charlie@example.com"));
            Console.WriteLine($"Created User: {created.Username} | ID: {created.Id}");

            await foreach (var user in userService.GetAllActiveUsersAsync())
            {
                Console.WriteLine($"Active User Stream: {user.Username} ({user.Email})");
            }
        }
    }
}

using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

public interface IDataRepository
{
    IAsyncEnumerable<int> GetStreamDataAsync(CancellationToken cancellationToken = default);
}

public class DataRepository : IDataRepository
{
    public async IAsyncEnumerable<int> GetStreamDataAsync(CancellationToken cancellationToken = default)
    {
        for (int i = 1; i <= 5; i++)
        {
            await Task.Delay(100, cancellationToken);
            yield return i * 100;
        }
    }
}

// Instruction: Implement an ASP.NET Core Middleware pattern in C# to log request execution time.
// Response:

using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Logging;
using System.Diagnostics;

public class RequestTimingMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<RequestTimingMiddleware> _logger;

    public RequestTimingMiddleware(RequestDelegate next, ILogger<RequestTimingMiddleware> logger)
    {
        _next = next;
        _logger = logger;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        var sw = Stopwatch.StartNew();
        try
        {
            await _next(context);
        }
        finally
        {
            sw.Stop();
            _logger.LogInformation("HTTP {Method} {Path} responded {StatusCode} in {Elapsed}ms",
                context.Request.Method, context.Request.Path, context.Response.StatusCode, sw.ElapsedMilliseconds);
        }
    }
}

// 2. Advanced Concurrent Pipeline & Channels (.NET 8)

using System.Threading.Channels;

public class DataProcessingPipeline
{
    private readonly Channel<int> _channel = Channel.CreateUnbounded<int>();

    public async Task ProduceAsync(IEnumerable<int> data)
    {
        foreach (var item in data)
        {
            await _channel.Writer.WriteAsync(item);
        }
        _channel.Writer.Complete();
    }

    public async Task ConsumeAsync(Func<int, Task> processAction)
    {
        await foreach (var item in _channel.Reader.ReadAllAsync())
        {
            await processAction(item);
        }
    }
}
