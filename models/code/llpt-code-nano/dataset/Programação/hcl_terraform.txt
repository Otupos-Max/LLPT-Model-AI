terraform {
  required_version = ">= 1.5.0"
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = var.aws_region
}

variable "aws_region" {
  type        = string
  default     = "us-east-1"
  description = "AWS target deployment region"
}

variable "environment" {
  type        = string
  default     = "prod"
  description = "Deployment environment name"
}

# VPC Network
resource "aws_vpc" "main" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true
  enable_dns_support   = true

  tags = {
    Name        = "vpc-llpt-${var.environment}"
    Environment = var.environment
  }
}

resource "aws_subnet" "public_a" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = "10.0.1.0/24"
  availability_zone       = "${var.aws_region}a"
  map_public_ip_on_launch = true

  tags = {
    Name = "subnet-public-a-${var.environment}"
  }
}

# ECS Cluster
resource "aws_ecs_cluster" "nano_cluster" {
  name = "ecs-cluster-llpt-${var.environment}"

  setting {
    name  = "containerInsights"
    value = "enabled"
  }
}

output "vpc_id" {
  value       = aws_vpc.main.id
  description = "Created VPC Identifier"
}

output "ecs_cluster_arn" {
  value       = aws_ecs_cluster.nano_cluster.arn
  description = "Amazon ECS Cluster ARN"
}
