export type UserStatus = "Active" | "Pending" | "Suspended"

export type User = {
    id: number,
    name: string,
    email: string,
    status: UserStatus,
    balance: number,
}

local UserModule = {}
UserModule.__index = UserModule

function UserModule.new(id: number, name: string, email: string): User
    local self: User = {
        id = id,
        name = name,
        email = email,
        status = "Active",
        balance = 0.0,
    }
    return self
end

function UserModule.credit(user: User, amount: number): number
    assert(amount > 0, "Credit amount must be positive")
    user.balance += amount
    return user.balance
end

function UserModule.debit(user: User, amount: number): (boolean, string?)
    if amount > user.balance then
        return false, "Insufficient balance"
    end
    user.balance -= amount
    return true, nil
end

-- Luau Parallel Luau & Task Library
local task = task or { spawn = function(f) coroutine.resume(coroutine.create(f)) end }

function UserModule.batchProcessUsers(users: { User })
    for _, usr in users do
        task.spawn(function()
            print("Processing user in Luau task:", usr.name, usr.email)
        end)
    end
end

return UserModule
