<#
.SYNOPSIS
    Automated system health and log auditor script.
.DESCRIPTION
    Collects system metrics, inspects disk space, and reports errors asynchronously.
#>
function Get-LLPTSystemMetrics {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $false)]
        [int]$DiskThresholdPercent = 15,

        [Parameter(Mandatory = $false)]
        [string]$LogOutputPath = "./system_report.json"
    )

    process {
        Write-Verbose "Collecting system drive metrics..."
        $disks = Get-CimInstance -ClassName Win32_LogicalDisk | Where-Object { $_.DriveType -eq 3 } | ForEach-Object {
            $freePercent = [math]::Round(($_.FreeSpace / $_.Size) * 100, 2)
            [PSCustomObject]@{
                DeviceID     = $_.DeviceID
                SizeGB       = [math]::Round($_.Size / 1GB, 2)
                FreeSpaceGB  = [math]::Round($_.FreeSpace / 1GB, 2)
                FreePercent  = $freePercent
                IsLowSpace   = $freePercent -lt $DiskThresholdPercent
            }
        }

        $report = [PSCustomObject]@{
            Timestamp   = (Get-Date).ToString("o")
            OSVersion   = (Get-CimInstance -ClassName Win32_OperatingSystem).Caption
            DiskReport  = $disks
        }

        $jsonOutput = $report | ConvertTo-Json -Depth 4
        if ($LogOutputPath) {
            $jsonOutput | Out-File -FilePath $LogOutputPath -Encoding utf8
        }

        return $report
    }
}

# Async REST Request Pipeline
$asyncJob = {
    param($url)
    try {
        $response = Invoke-RestMethod -Uri $url -Method Get -TimeoutSec 10
        return $response
    } catch {
        Write-Error "Failed to query API: $_"
    }
}
