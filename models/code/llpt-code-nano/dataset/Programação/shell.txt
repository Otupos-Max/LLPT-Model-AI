#!/usr/bin/env bash
# Shell Dataset - Modern Robust Bash Scripting, Automation & Systems Management

set -euo pipefail
IFS=$'\n\t'

# ==========================================
# 1. Colors & Logging Utility Functions
# ==========================================

readonly RED='\033[0;31m'
readonly GREEN='\033[0;32m'
readonly YELLOW='\033[1;33m'
readonly BLUE='\033[0;34m'
readonly NC='\033[0m' # No Color

log_info() {
    echo -e "${GREEN}[INFO]$(date +'%Y-%m-%dT%H:%M:%S%z'):${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]$(date +'%Y-%m-%dT%H:%M:%S%z'):${NC} $1" >&2
}

log_error() {
    echo -e "${RED}[ERROR]$(date +'%Y-%m-%dT%H:%M:%S%z'):${NC} $1" >&2
}

# Cleanup on exit
cleanup() {
    local exit_code=$?
    if [ $exit_code -ne 0 ]; then
        log_error "Script failed with exit code ${exit_code}."
    else
        log_info "Script completed successfully."
    fi
    # Remove temporary directory if exists
    if [ -n "${TMP_DIR:-}" ] && [ -d "${TMP_DIR}" ]; then
        rm -rf "${TMP_DIR}"
    fi
}
trap cleanup EXIT

# ==========================================
# 2. Dependency Checks & Helper Functions
# ==========================================

check_command() {
    local cmd=$1
    if ! command -v "${cmd}" &> /dev/null; then
        log_error "Required command '${cmd}' is not installed or not in PATH."
        exit 1
    fi
}

check_command "curl"
check_command "jq"
check_command "tar"

# ==========================================
# 3. CLI Option Parsing (getopts)
# ==========================================

usage() {
    cat <<EOF
Usage: $(basename "$0") [OPTIONS]

Options:
  -d, --directory DIR    Target directory to backup (Required)
  -o, --output DIR       Backup destination folder (Default: /tmp/backups)
  -v, --verbose          Enable verbose output
  -h, --help             Show this help message and exit
EOF
    exit 1
}

TARGET_DIR=""
OUTPUT_DIR="/tmp/backups"
VERBOSE=false

while [[ $# -gt 0 ]]; do
    case $1 in
        -d|--directory)
            TARGET_DIR="$2"
            shift 2
            ;;
        -o|--output)
            OUTPUT_DIR="$2"
            shift 2
            ;;
        -v|--verbose)
            VERBOSE=true
            shift
            ;;
        -h|--help)
            usage
            ;;
        *)
            log_error "Unknown argument: $1"
            usage
            ;;
    esac
done

if [ -z "${TARGET_DIR}" ]; then
    log_error "Missing required argument: --directory"
    usage
fi

# ==========================================
# 4. Backup Operation
# ==========================================

perform_backup() {
    local src=$1
    local dest=$2

    if [ ! -d "${src}" ]; then
        log_error "Source directory '${src}' does not exist."
        exit 1
    fi

    mkdir -p "${dest}"
    TMP_DIR=$(mktemp -d)

    local timestamp
    timestamp=$(date +'%Y%m%d_%H%M%S')
    local archive_name="backup_${timestamp}.tar.gz"
    local archive_path="${dest}/${archive_name}"

    log_info "Creating archive '${archive_path}' from '${src}'..."

    if [ "${VERBOSE}" = true ]; then
        tar -czvf "${archive_path}" -C "$(dirname "${src}")" "$(basename "${src}")"
    else
        tar -czf "${archive_path}" -C "$(dirname "${src}")" "$(basename "${src}")"
    fi

    log_info "Backup archive created successfully at '${archive_path}'."
}

perform_backup "${TARGET_DIR}" "${OUTPUT_DIR}"
