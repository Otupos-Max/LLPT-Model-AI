package com.llpt.ui {
    import flash.display.Sprite;
    import flash.events.Event;
    import flash.events.MouseEvent;
    import flash.text.TextField;
    import flash.text.TextFormat;

    public class CustomButton extends Sprite {
        private var label:TextField;
        private var isPressed:Boolean = false;

        public function CustomButton(buttonText:String) {
            initBackground();
            initLabel(buttonText);
            addEventListener(MouseEvent.CLICK, onClickHandler);
        }

        private function initBackground():void {
            graphics.clear();
            graphics.beginFill(isPressed ? 0x0055A5 : 0x007ACC);
            graphics.drawRoundRect(0, 0, 140, 45, 12, 12);
            graphics.endFill();
        }

        private function initLabel(textStr:String):void {
            label = new TextField();
            label.text = textStr;
            label.x = 15;
            label.y = 12;
            label.selectable = false;

            var tf:TextFormat = new TextFormat("Arial", 14, 0xFFFFFF, true);
            label.setTextFormat(tf);
            addChild(label);
        }

        private function onClickHandler(event:MouseEvent):void {
            isPressed = !isPressed;
            initBackground();
            dispatchEvent(new Event("buttonStateChanged", true));
        }
    }
}
