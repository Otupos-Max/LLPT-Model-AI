export interface User {
  readonly id: string;
  name: string;
  email: string;
  role: 'admin' | 'user' | 'guest';
  createdAt: Date;
}

export type ApiResponse<T> =
  | { success: true; data: T; timestamp: number }
  | { success: false; error: string; code: number };

export class HttpClient {
  private baseUrl: string;

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl.replace(/\/$/, '');
  }

  async get<T>(endpoint: string, headers?: Record<string, string>): Promise<ApiResponse<T>> {
    try {
      const response = await fetch(`${this.baseUrl}${endpoint}`, {
        method: 'GET',
        headers: { 'Content-Type': 'application/json', ...headers }
      });
      if (!response.ok) {
        return { success: false, error: response.statusText, code: response.status };
      }
      const data: T = await response.json();
      return { success: true, data, timestamp: Date.now() };
    } catch (err) {
      return { success: false, error: (err as Error).message, code: 500 };
    }
  }
}

// Advanced TypeScript generic repository interface
export interface Repository<T extends { id: string }> {
  findById(id: string): Promise<T | null>;
  findAll(): Promise<T[]>;
  save(item: Omit<T, 'id'>): Promise<T>;
  delete(id: string): Promise<boolean>;
}

export class InMemoryRepository<T extends { id: string }> implements Repository<T> {
  private items = new Map<string, T>();

  async findById(id: string): Promise<T | null> {
    return this.items.get(id) ?? null;
  }

  async findAll(): Promise<T[]> {
    return Array.from(this.items.values());
  }

  async save(item: Omit<T, 'id'>): Promise<T> {
    const id = Math.random().toString(36).substr(2, 9);
    const newItem = { ...item, id } as T;
    this.items.set(id, newItem);
    return newItem;
  }

  async delete(id: string): Promise<boolean> {
    return this.items.delete(id);
  }
}

type IsString<T> = T extends string ? true : false;

type DeepReadonly<T> = {
    readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

interface UserProfile {
    id: number;
    details: {
        name: string;
        email: string;
    };
}

type ReadonlyUserProfile = DeepReadonly<UserProfile>;
