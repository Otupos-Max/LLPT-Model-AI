program CustomerManagement;

{$MODE OBJFPC}{$H+}{$M+}

uses
  SysUtils, Classes;

type
  EInvalidDataException = class(Exception);

  TCustomerStatus = (csActive, csPending, csSuspended);

  TCustomer = class
  private
    FId: Integer;
    FName: string;
    FBalance: Double;
    FStatus: TCustomerStatus;
    procedure SetBalance(const Value: Double);
  public
    constructor Create(AId: Integer; AName: string);
    destructor Destroy; override;
    function ToString: string; override;
    property Id: Integer read FId;
    property Name: string read FName write FName;
    property Balance: Double read FBalance write SetBalance;
    property Status: TCustomerStatus read FStatus write FStatus;
  end;

constructor TCustomer.Create(AId: Integer; AName: string);
begin
  inherited Create;
  FId := AId;
  FName := AName;
  FBalance := 0.0;
  FStatus := csActive;
end;

destructor TCustomer.Destroy;
begin
  inherited Destroy;
end;

procedure TCustomer.SetBalance(const Value: Double);
begin
  if Value < 0 then
    raise EInvalidDataException.Create('Balance cannot be negative');
  FBalance := Value;
end;

function TCustomer.ToString: string;
begin
  Result := Format('Customer ID: %d | Name: %s | Balance: %.2f', [FId, FName, FBalance]);
end;

var
  Cust: TCustomer;
begin
  try
    Cust := TCustomer.Create(101, 'Pascal Developer');
    try
      Cust.Balance := 250.50;
      WriteLn(Cust.ToString);
    finally
      Cust.Free;
    end;
  except
    on E: Exception do
      WriteLn('Error: ', E.Message);
  end;
end.
