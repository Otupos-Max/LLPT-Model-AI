namespace LLPT.CodeNano

open System
open System.Threading.Tasks

// Discriminated Unions & Algebraic Data Types
type Result<'T, 'E> =
    | Ok of 'T
    | Error of 'E

type PaymentMethod =
    | CreditCard of cardNumber: string * expiry: string
    | PayPal of email: string
    | Crypto of walletAddress: string

type OrderState =
    | Pending
    | Processing of payment: PaymentMethod
    | Completed of transactionId: Guid
    | Failed of reason: string

// Record Type with Immutability
type Customer = {
    Id: Guid
    Name: string
    Balance: decimal
    State: OrderState
}

module CustomerOperations =

    let updateBalance (customer: Customer) (amount: decimal) : Customer =
        { customer with Balance = customer.Balance + amount }

    let processPayment (customer: Customer) (method: PaymentMethod) : Customer =
        match method with
        | CreditCard(card, _) when card.StartsWith("4") ->
            { customer with State = Completed (Guid.NewGuid()) }
        | PayPal email when email.Contains("@") ->
            { customer with State = Completed (Guid.NewGuid()) }
        | Crypto wallet when wallet.Length > 20 ->
            { customer with State = Completed (Guid.NewGuid()) }
        | _ ->
            { customer with State = Failed "Invalid payment parameters" }

    // Async & Pipeline Processing
    let fetchCustomerAsync (id: Guid) : Async<Result<Customer, string>> = async {
        do! Async.Sleep 100
        if id = Guid.Empty then
            return Error "Customer ID cannot be empty"
        else
            let customer = { Id = id; Name = "John Doe"; Balance = 500.00m; State = Pending }
            return Ok customer
    }

// Active Patterns
let (|Integer|Float|String|) (input: string) =
    match Int32.TryParse input, Double.TryParse input with
    | (true, intVal), _ -> Integer intVal
    | _, (true, floatVal) -> Float floatVal
    | _ -> String input
