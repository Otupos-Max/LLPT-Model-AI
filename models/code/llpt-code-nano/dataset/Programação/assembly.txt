;; ==========================================
;; LLPT Code Nano - Assembly (x86_64 NASM & ARM64 AAPCS)
;; ==========================================

global _start

section .data
    msg db "Hello, LLPT Code Nano Assembly Engine!", 0xA
    len equ $ - msg
    num_array dq 10, 20, 30, 40, 50
    array_size equ 5

section .bss
    sum_res resq 1

section .text

;; NASM x86_64 Syscall - Write & Exit
_start:
    ;; Calculate sum of num_array
    mov rsi, num_array
    mov rcx, array_size
    xor rax, rax        ;; accumulator = 0

sum_loop:
    add rax, [rsi]
    add rsi, 8
    loop sum_loop

    mov [sum_res], rax  ;; Store calculated sum

    ;; Write message to stdout
    mov rax, 1          ;; sys_write
    mov rdi, 1          ;; stdout
    mov rsi, msg        ;; buffer
    mov rdx, len        ;; count
    syscall

    ;; Exit program
    mov rax, 60         ;; sys_exit
    xor rdi, rdi        ;; status code 0
    syscall

;; Function: vector_add_avx2 (C calling convention: RDI = a, RSI = b, RDX = result, RCX = count)
global vector_add_avx2
vector_add_avx2:
    push rbp
    mov rbp, rsp
    xor rax, rax

.loop:
    cmp rax, rcx
    jge .done
    vmovups ymm0, [rdi + rax*4]
    vmovups ymm1, [rsi + rax*4]
    vaddps  ymm2, ymm0, ymm1
    vmovups [rdx + rax*4], ymm2
    add rax, 8
    jmp .loop

.done:
    pop rbp
    ret
