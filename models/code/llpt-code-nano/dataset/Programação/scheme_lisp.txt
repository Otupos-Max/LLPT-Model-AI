;; Higher order pipelines & continuation-passing style (CPS)
(define (filter-map proc lst)
  (let loop ((rest lst) (acc '()))
    (if (null? rest)
        (reverse acc)
        (let ((res (proc (car rest))))
          (if res
              (loop (cdr rest) (cons res acc))
              (loop (cdr rest) acc))))))

;; Tail-Call Optimized Fibonacci
(define (fibonacci n)
  (define (fib-iter a b count)
    (if (= count 0)
        a
        (fib-iter b (+ a b) (- count 1))))
  (fib-iter 0 1 n))

;; Hygienic Syntax-Rules Macro
(define-syntax-rule (when condition body ...)
  (if condition
      (begin body ...)))

;; Call-with-current-continuation (call/cc) early exit pattern
(define (find-first-negative lst)
  (call/cc
   (lambda (exit)
     (for-each (lambda (x)
                 (when (< x 0)
                   (exit x)))
               lst)
     #f)))
