syntax = "proto3";

package llpt.code.nano.v1;

option go_package = "github.com/llpt/code/nano/v1;nanov1";
option java_multiple_files = true;
option java_package = "com.llpt.code.nano.v1";

enum UserRole {
  USER_ROLE_UNSPECIFIED = 0;
  USER_ROLE_GUEST = 1;
  USER_ROLE_MEMBER = 2;
  USER_ROLE_ADMIN = 3;
}

message User {
  int64 id = 1;
  string username = 2;
  string email = 3;
  UserRole role = 4;
  double balance = 5;
  repeated string tags = 6;
  map<string, string> metadata = 7;
}

message CreateUserRequest {
  string username = 1;
  string email = 2;
  UserRole role = 3;
}

message CreateUserResponse {
  User user = 1;
}

message GetUserRequest {
  int64 id = 1;
}

message StreamUserEventsRequest {
  int64 user_id = 1;
}

message UserEvent {
  int64 event_id = 1;
  string event_type = 2;
  string payload_json = 3;
}

// gRPC Service Definition
service UserService {
  rpc CreateUser(CreateUserRequest) returns (CreateUserResponse);
  rpc GetUser(GetUserRequest) returns (User);
  rpc StreamEvents(StreamUserEventsRequest) returns (stream UserEvent);
}
