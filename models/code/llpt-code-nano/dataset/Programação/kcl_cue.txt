# ------------------------------------------
# CUE (Configure, Unify, Execute)
# ------------------------------------------
#Schema: {
    name: string
    port: int & >=1024 & <=65535
    env:  "dev" | "staging" | "prod"
    replicas: int & >0
    db: {
        host: string
        port: int | *5432
    }
}

appConfig: #Schema & {
    name:     "llpt-code-nano"
    port:     8080
    env:      "prod"
    replicas: 3
    db: {
        host: "postgres.internal"
    }
}

# ------------------------------------------
# KCL (Kusion Configuration Language)
# ------------------------------------------
schema Container:
    name: str
    image: str
    command?: [str]
    env?: {str: str}

schema Deployment:
    apiVersion: str = "apps/v1"
    kind: str = "Deployment"
    name: str
    replicas: int = 1
    containers: [Container]

myDeployment = Deployment {
    name = "llpt-nano"
    replicas = 3
    containers = [
        Container {
            name = "nano-api"
            image = "llpt/code-nano:1.2.0"
            env = {
                "PORT": "8080"
                "ENV": "production"
            }
        }
    ]
}
