<?php
// PHP Dataset - Modern PHP 8+ Architecture, Attributes, PDO & Design Patterns

namespace LLPT\CodeDataset;

use PDO;
use PDOException;
use InvalidArgumentException;
use RuntimeException;
use DateTimeImmutable;

// ==========================================
// 1. PHP 8 Backed Enums & Readonly Data Transfer Objects
// ==========================================

enum UserRole: string {
    case Admin = 'admin';
    case Developer = 'developer';
    case Guest = 'guest';

    public function label(): string {
        return match($this) {
            self::Admin => 'Administrator',
            self::Developer => 'Software Engineer',
            self::Guest => 'Guest Visitor',
        };
    }
}

readonly class UserDTO {
    public function __construct(
        public string $id,
        public string $username,
        public string $email,
        public UserRole $role,
        public bool $isActive,
        public DateTimeImmutable $createdAt
    ) {
        if (!filter_var($this->email, FILTER_VALIDATE_EMAIL)) {
            throw new InvalidArgumentException("Invalid email format: {$this->email}");
        }
    }

    public function toArray(): array {
        return [
            'id' => $this->id,
            'username' => $this->username,
            'email' => $this->email,
            'role' => $this->role->value,
            'is_active' => $this->isActive,
            'created_at' => $this->createdAt->format(DATE_ATOM)
        ];
    }
}

// ==========================================
// 2. PDO Database Repository Pattern
// ==========================================

interface UserRepositoryInterface {
    public function findById(string $id): ?UserDTO;
    public function save(UserDTO $user): bool;
    public function findActiveByRole(UserRole $role): array;
}

class PdoUserRepository implements UserRepositoryInterface {
    public function __construct(
        private PDO $pdo
    ) {}

    public function findById(string $id): ?UserDTO {
        $stmt = $this->pdo->prepare("
            SELECT id, username, email, role, is_active, created_at
            FROM users
            WHERE id = :id
            LIMIT 1
        ");
        $stmt->execute(['id' => $id]);
        $data = $stmt->fetch(PDO::FETCH_ASSOC);

        if (!$data) {
            return null;
        }

        return new UserDTO(
            id: $data['id'],
            username: $data['username'],
            email: $data['email'],
            role: UserRole::from($data['role']),
            isActive: (bool)$data['is_active'],
            createdAt: new DateTimeImmutable($data['created_at'])
        );
    }

    public function save(UserDTO $user): bool {
        $stmt = $this->pdo->prepare("
            INSERT INTO users (id, username, email, role, is_active, created_at)
            VALUES (:id, :username, :email, :role, :is_active, :created_at)
            ON DUPLICATE KEY UPDATE
                username = VALUES(username),
                email = VALUES(email),
                role = VALUES(role),
                is_active = VALUES(is_active)
        ");

        return $stmt->execute([
            'id' => $user->id,
            'username' => $user->username,
            'email' => $user->email,
            'role' => $user->role->value,
            'is_active' => (int)$user->isActive,
            'created_at' => $user->createdAt->format('Y-m-d H:i:s')
        ]);
    }

    public function findActiveByRole(UserRole $role): array {
        $stmt = $this->pdo->prepare("
            SELECT id, username, email, role, is_active, created_at
            FROM users
            WHERE role = :role AND is_active = 1
            ORDER BY username ASC
        ");
        $stmt->execute(['role' => $role->value]);
        $rows = $stmt->fetchAll(PDO::FETCH_ASSOC);

        return array_map(fn($data) => new UserDTO(
            id: $data['id'],
            username: $data['username'],
            email: $data['email'],
            role: UserRole::from($data['role']),
            isActive: true,
            createdAt: new DateTimeImmutable($data['created_at'])
        ), $rows);
    }
}

// ==========================================
// 3. Service Layer with Exception Handling
// ==========================================

class UserService {
    public function __construct(
        private UserRepositoryInterface $repository
    ) {}

    public function registerUser(string $username, string $email, UserRole $role): UserDTO {
        $id = bin2hex(random_bytes(8));
        $user = new UserDTO(
            id: $id,
            username: $username,
            email: $email,
            role: $role,
            isActive: true,
            createdAt: new DateTimeImmutable()
        );

        if (!$this->repository->save($user)) {
            throw new RuntimeException("Failed to persist user in database.");
        }

        return $user;
    }
}


// ==========================================
// 4. PSR-11 Container & Middleware Pattern
// ==========================================

namespace App\Core;

interface ContainerInterface {
    public function get(string $id);
    public function has(string $id): bool;
}

class ServiceContainer implements ContainerInterface {
    private array $services = [];

    public function set(string $id, callable $factory): void {
        $this->services[$id] = $factory;
    }

    public function get(string $id) {
        if (!$this->has($id)) {
            throw new \Exception("Service not found: {$id}");
        }
        return $this->services[$id]($this);
    }

    public function has(string $id): bool {
        return isset($this->services[$id]);
    }
}
