program ObjectPascalDemo;

type
  TArrayOfInt = array of Integer;

function CalculateSum(const Arr: TArrayOfInt): Integer;
var
  i, Total: Integer;
begin
  Total := 0;
  for i := Low(Arr) to High(Arr) do
    Total := Total + Arr[i];
  Result := Total;
end;

procedure ProcessData;
var
  Data: TArrayOfInt;
  Res: Integer;
begin
  SetLength(Data, 3);
  Data[0] := 10;
  Data[1] := 20;
  Data[2] := 30;
  Res := CalculateSum(Data);
  Writeln('Sum result: ', Res);
end;

begin
  ProcessData;
end.
