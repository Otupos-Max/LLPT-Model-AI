(module
  ;; Memory declaration (1 page = 64KB)
  (memory (export "memory") 1)

  ;; Exported function: Vector Dot Product (f32)
  (func (export "dot_product_f32")
        (param $a_ptr i32)
        (param $b_ptr i32)
        (param $len i32)
        (result f32)
    (local $i i32)
    (local $acc f32)

    (local.set $i (i32.const 0))
    (local.set $acc (f32.const 0.0))

    (block $break
      (loop $top
        (br_if $break (i32.ge_s (local.get $i) (local.get $len)))

        ;; acc += a[i] * b[i]
        (local.set $acc
          (f32.add
            (local.get $acc)
            (f32.mul
              (f32.load (i32.add (local.get $a_ptr) (i32.shl (local.get $i) (i32.const 2))))
              (f32.load (i32.add (local.get $b_ptr) (i32.shl (local.get $i) (i32.const 2))))
            )
          )
        )

        (local.set $i (i32.add (local.get $i) (i32.const 1)))
        (br $top)
      )
    )
    (local.get $acc)
  )
)
