# ==============================================================================
# 1. Dockerfile Multi-Stage Build Otimizado para Produção Python / FastAPI
# ==============================================================================

FROM python:3.12-slim AS builder

WORKDIR /app

ENV PYTHONUNBUFFERED=1 \
    PYTHONDONTWRITEBYTECODE=1

RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
RUN pip install --no-cache-dir --prefix=/install -r requirements.txt

FROM python:3.12-slim AS runner

WORKDIR /app

RUN useradd -m -u 1000 appuser

COPY --from=builder /install /usr/local
COPY . /app

USER appuser

EXPOSE 8000

CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000"]


# ==============================================================================
# 2. Prevenção de SQL Injection (OWASP Top 10) em Python/PostgreSQL
# ==============================================================================

def get_user_secure(cursor, username: str):
    """Executes safe parameterized SQL query preventing SQL injection."""
    query = "SELECT user_id, username, email, is_active FROM users WHERE username = %s LIMIT 1"
    cursor.execute(query, (username,))
    return cursor.fetchone()

# ==========================================
# Dockerfile Multi-stage Build & Linux Hardening

# Stage 1: Build
FROM golang:1.22-alpine AS builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -o app .

# Stage 2: Minimal Distroless Security Execution
FROM gcr.io/distroless/static-debian12
COPY --from=builder /app/app /app
USER nonroot:nonroot
ENTRYPOINT ["/app"]
