# CMakeLists.txt (Modern CMake 3.20+)

cmake_minimum_required(VERSION 3.20)
project(LLPTCodeNanoEngine VERSION 1.2.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Options
option(BUILD_SHARED_LIBS "Build shared library" OFF)
option(ENABLE_AVX2 "Enable AVX2 vector optimization" ON)

if(ENABLE_AVX2 AND (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang"))
    add_compile_options(-mavx2 -mfma -O3)
endif()

add_library(llpt_nano_core STATIC
    src/core/model.cpp
    src/core/tokenizer.cpp
    src/core/tensor.cpp
)

target_include_directories(llpt_nano_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

find_package(Threads REQUIRED)
target_link_libraries(llpt_nano_core PRIVATE Threads::Threads)

# Executable CLI
add_executable(llpt_cli src/main.cpp)
target_link_libraries(llpt_cli PRIVATE llpt_nano_core)
