(* Example 2: Recursive Data Structures in Idris 2 *)
let rec factorial n =
    if n <= 1 then 1
    else n * factorial (n - 1)

(* ==========================================
   2. Fill-In-The-Middle (FIM) Example
   ========================================== *)
<PRE>
let rec sum_list lst =
    match lst with
    | [] -> 0
<SUF>
<MID>
    | x :: xs -> x + sum_list xs
<EOT>
