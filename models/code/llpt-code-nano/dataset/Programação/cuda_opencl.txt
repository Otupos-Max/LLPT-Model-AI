// CUDA Kernel: Vector Addition with Grid-Stride Loop
__global__ void vector_add_kernel(const float* A, const float* B, float* C, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = blockDim.x * gridDim.x;

    for (int i = idx; i < N; i += stride) {
        C[i] = A[i] + B[i];
    }
}

// CUDA Kernel: Parallel Matrix Multiplication with Shared Memory Tiling
#define TILE_WIDTH 16

__global__ void matrix_multiply_tiled(const float* A, const float* B, float* C, int width) {
    __shared__ float s_A[TILE_WIDTH][TILE_WIDTH];
    __shared__ float s_B[TILE_WIDTH][TILE_WIDTH];

    int bx = blockIdx.x;  int by = blockIdx.y;
    int tx = threadIdx.x; int ty = threadIdx.y;

    int Row = by * TILE_WIDTH + ty;
    int Col = bx * TILE_WIDTH + tx;

    float Pvalue = 0.0f;

    for (int ph = 0; ph < width / TILE_WIDTH; ++ph) {
        s_A[ty][tx] = A[Row * width + ph * TILE_WIDTH + tx];
        s_B[ty][tx] = B[(ph * TILE_WIDTH + ty) * width + Col];
        __syncthreads();

        for (int k = 0; k < TILE_WIDTH; ++k) {
            Pvalue += s_A[ty][k] * s_B[k][tx];
        }
        __syncthreads();
    }
    if (Row < width && Col < width) {
        C[Row * width + Col] = Pvalue;
    }
}
