{-# LANGUAGE DeriveGeneric #-}
{-# LANGUAGE OverloadedStrings #-}
{-# LANGUAGE FlexibleContexts #-}

module Main where

import GHC.Generics
import Data.Text (Text)
import qualified Data.Text as T
import Control.Monad.State
import Control.Monad.Except

-- Algebraic Data Types & GADTs
data PaymentStatus
  = Pending
  | Processing
  | Succeeded Text
  | Failed Text
  deriving (Show, Eq, Generic)

data User = User
  { userId    :: Int
  , userName  :: Text
  , userEmail :: Text
  } deriving (Show, Eq, Generic)

-- Monad Transformer Stack
type AppState = [User]
type AppMonad a = ExceptT Text (State AppState) a

registerUser :: Text -> Text -> AppMonad User
registerUser name email = do
  users <- get
  let newId = length users + 1
  if any (\u -> userEmail u == email) users
    then throwError "User with this email already exists"
    else do
      let newUser = User newId name email
      put (newUser : users)
      return newUser

-- Higher Order Pipelines & QuickCheck
quickSort :: Ord a => [a] -> [a]
quickSort [] = []
quickSort (x:xs) =
  let smaller = quickSort [a | a <- xs, a <= x]
      larger  = quickSort [b | b <- xs, b > x]
  in smaller ++ [x] ++ larger

main :: IO ()
main = do
  let initialUsers = []
      action = do
        u1 <- registerUser "Alice" "alice@example.com"
        u2 <- registerUser "Bob" "bob@example.com"
        return [u1, u2]
      (result, finalState) = runState (runExceptT action) initialUsers
  case result of
    Right users -> putStrLn $ "Registered users: " ++ show (length users)
    Left err    -> putStrLn $ "Error: " ++ T.unpack err
