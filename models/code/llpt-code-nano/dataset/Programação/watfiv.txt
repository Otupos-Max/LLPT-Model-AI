section .text
global _start

_start:
    mov eax, 1        ; system call for exit
    mov ebx, 0        ; return status 0
    int 0x80          ; execute syscall

; ==========================================
; 2. Fill-In-The-Middle (FIM) Example
; ==========================================
<PRE>
add_numbers:
    mov eax, [esp+4]
<SUF>
    ret
<MID>
    add eax, [esp+8]
<EOT>
