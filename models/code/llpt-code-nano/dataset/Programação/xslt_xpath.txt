<!-- ==========================================
     LLPT Code Nano - XSLT 3.0 & XPath/XQuery Transformations
     ========================================== -->

<xsl:stylesheet version="3.0"
                xmlns:xsl="http://www.w3.org/1999/XSL/Transform"
                xmlns:xs="http://www.w3.org/2001/XMLSchema"
                xmlns:llpt="http://llpt.dev/functions"
                exclude-result-prefixes="xs llpt">

    <xsl:output method="html" version="5.0" encoding="UTF-8" indent="yes"/>

    <!-- Custom XSLT Function -->
    <xsl:function name="llpt:calculate-tier" as="xs:string">
        <xsl:param name="revenue" as="xs:decimal"/>
        <xsl:choose>
            <xsl:when test="$revenue >= 100000.0">Gold Tier</xsl:when>
            <xsl:when test="$revenue >= 25000.0">Silver Tier</xsl:when>
            <xsl:otherwise>Standard Tier</xsl:otherwise>
        </xsl:choose>
    </xsl:function>

    <!-- Template Matching Root -->
    <xsl:template match="/customers">
        <html>
            <head>
                <title>LLPT Customer Audit Report</title>
                <style>
                    table { border-collapse: collapse; width: 100%; }
                    th, td { border: 1px solid #ccc; padding: 8px; text-align: left; }
                    th { background-color: #f2f2f2; }
                </style>
            </head>
            <body>
                <h1>Customer Revenue Summary</h1>
                <table>
                    <thead>
                        <tr>
                            <th>ID</th>
                            <th>Name</th>
                            <th>Revenue</th>
                            <th>Tier</th>
                        </tr>
                    </thead>
                    <tbody>
                        <xsl:apply-templates select="customer[status = 'Active']">
                            <xsl:sort select="revenue" data-type="number" order="descending"/>
                        </xsl:apply-templates>
                    </tbody>
                </table>
            </body>
        </html>
    </xsl:template>

    <xsl:template match="customer">
        <tr>
            <td><xsl:value-of select="@id"/></td>
            <td><xsl:value-of select="name"/></td>
            <td>$<xsl:value-of select="format-number(revenue, '#,##0.00')"/></td>
            <td><xsl:value-of select="llpt:calculate-tier(xs:decimal(revenue))"/></td>
        </tr>
    </xsl:template>

</xsl:stylesheet>
